    72→function useDebouncedValue<T>(value: T, delayMs: number): T {
    73→  const [debounced, setDebounced] = useState(value)
    74→  useEffect(() => {
    75→    const t = setTimeout(() => setDebounced(value), delayMs)
    76→    return () => clearTimeout(t)
    77→  }, [value, delayMs])
    78→  return debounced
    79→}

    81→function renderCellValue(value: unknown, type: string | undefined): React.ReactNode {
    82→  if (value === null || value === undefined || value === '') {
    83→    return <span className="text-muted-foreground/60">∅</span>
    84→  }
    85→  if (value === MASK_LITERAL) {
    86→    return <span className="italic text-muted-foreground/80">{MASK_LITERAL}</span>
    87→  }
    88→  if (typeof value === 'boolean' || type === 'boolean') {
    89→    return (
    90→      <span className={value ? 'text-amber-600 dark:text-amber-400' : 'text-muted-foreground'}>
    91→        {String(value)}
    92→      </span>
    93→    )
    94→  }
    95→  if (typeof value === 'number') {
    96→    return <span className="tabular-nums">{value.toLocaleString()}</span>
    97→  }
    98→  return String(value)
    99→}

   525→                              col.type === 'number' && 'text-right tabular-nums',
   526→                            )}
   527→                          >
   528→                            {renderCellValue(row[col.name], col.type)}
   529→                          </TableCell>
   530→                        ))}
   531→                      </TableRow>
   532→                    ))}
   533→                  </TableBody>
   534→                </Table>
   535→              </div>
   536→            )}
   537→
   538→            {/* Pagination footer */}
   539→            <div className="flex items-center justify-between gap-2 border-t border-border p-3">
   540→              <Button
   541→                variant="outline"
   542→                size="sm"
   543→                disabled={page <= 1 || rowsQuery.isFetching}

   548→                <ChevronLeft className="h-3.5 w-3.5" aria-hidden />

   598→            </CardContent>
   599→          </Card>
   600→        </TabsContent>
   601→
   602→        {/* ── History ── */}
   603→        <TabsContent value="history" className="mt-3">
   604→          {/* R14: version compare (schema + row diff) */}
   605→          <div className="mb-4">
   606→            <DatasetDiffPanel
   607→              datasetId={id}
   608→              versions={detail.versions.map((v) => ({ versionNumber: v.versionNumber, rowCount: v.rowCount }))}
   609→            />
   610→          </div>
   611→
   612→          {/* Retention (M2): keepLastN select + manual prune */}
   613→          <div className="mb-4 flex flex-col gap-3 rounded-lg border border-border bg-muted/40 p-3 sm:flex-row sm:items-center sm:justify-between">
   614→            <div className="flex min-w-0 items-start gap-2.5">
   615→              <Archive className="mt-0.5 h-4 w-4 shrink-0 text-muted-foreground" aria-hidden />
   616→              <div className="min-w-0">
   617→                <p className="text-xs font-medium text-foreground">
   618→                  Version retention{' '}

   629→              </div>
   630→            </div>
   631→            <div className="flex shrink-0 items-center gap-2">

   670→                Prune now

   672→            </div>
   673→          </div>
   674→
   675→          <Card>
   676→            <CardContent className="p-4">
   677→              <ol className="relative ml-3 space-y-4 border-l border-border pl-6">
   678→                {detail.versions.map((v, idx) => {
   679→                  const isLatest = idx === 0
   680→                  const isSelected = v.versionNumber === version

   685→                        className={cn(

   693→                        className={cn(

   743→          </Card>
   744→        </TabsContent>
   745→
   746→        {/* ── Quality (R14) ── */}
