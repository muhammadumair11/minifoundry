   393→   393→// ── API usage mini-dashboard (Task 7-c) ──────────────────────────────
   394→   394→// Aggregates the token list client-side: total requests, active count,
   395→   395→// most-used active token, per-token emerald bars (width ∝ requestCount vs
   396→   396→// max) and a dimmed "retired" group for revoked tokens.
   397→   397→
   398→   398→function ApiUsageCard({ tokens, loading }: { tokens: ApiTokenInfo[]; loading: boolean }) {
   399→   399→  const active = tokens
   400→   400→    .filter((t) => !t.revokedAt)
   401→   401→    .sort((a, b) => b.requestCount - a.requestCount)
   402→   402→  const retired = tokens.filter((t) => t.revokedAt)
   403→   403→  const totalRequests = tokens.reduce((n, t) => n + t.requestCount, 0)
   404→   404→  const maxRequests = active.length > 0 ? Math.max(...active.map((t) => t.requestCount)) : 0
   405→   405→  const top = active.find((t) => t.requestCount > 0)
   406→   406→  const hasTraffic = totalRequests > 0
   407→   407→
   408→   408→  return (
   409→   409→    <Card className="mb-6 border-zinc-800 bg-zinc-900/60">
   410→   410→      <CardHeader className="gap-1.5 border-b border-zinc-800 p-4 sm:p-5">
   411→   411→        <CardTitle className="flex items-center gap-2 text-sm font-semibold text-zinc-200">
   412→   412→          <Activity className="h-4 w-4 text-emerald-400" aria-hidden />
   413→   413→          API usage
   414→   414→        </CardTitle>
   415→   415→        <CardDescription className="mt-1 text-xs text-zinc-500">
   416→   416→          Live traffic across all tokens — every authenticated{' '}
   417→   417→          <span className="font-mono text-zinc-400">/api/foundry/v1</span> call counts here.
   418→   418→        </CardDescription>
   419→   419→      </CardHeader>
   420→   420→      <CardContent className="p-4 sm:p-5">
   421→   421→        {loading ? (
   422→   422→          <div className="space-y-3">
   423→   423→            <div className="grid grid-cols-1 gap-3 sm:grid-cols-3">
   424→   424→              {[0, 1, 2].map((i) => (
   425→   425→                <Skeleton key={i} className="h-10 w-full bg-zinc-800/70" />
   426→   426→              ))}
   427→   427→            </div>
   428→   428→            <Skeleton className="h-16 w-full bg-zinc-800/70" />
   429→   429→          </div>
   430→   430→        ) : tokens.length === 0 ? (
   431→   431→          <p className="py-2 text-center text-sm text-zinc-500">
   432→   432→            No tokens yet — create one below to start calling the read API.
   433→   433→          </p>

   540→   540→function CreateTokenDialog({
   541→   541→  open,
   542→   542→  onOpenChange,
   543→   543→}: {
   544→   544→  open: boolean
   545→   545→  onOpenChange: (open: boolean) => void
   546→   546→}) {
   547→   547→  const queryClient = useQueryClient()
   548→   548→  const [name, setName] = useState('')
   549→   549→  const [created, setCreated] = useState<CreateTokenResponse | null>(null)
   550→   550→
   551→   551→  const trimmed = name.trim()
   552→   552→  const nameValid = trimmed.length >= 1 && trimmed.length <= 64
   553→   553→
   554→   554→  const createMutation = useMutation({
   555→   555→    mutationFn: () => apiSend<CreateTokenResponse>('/api/tokens', 'POST', { name: trimmed }),
   556→   556→    onSuccess: (res) => {
   557→   557→      // Remember locally for the Explorer before anything else can unmount.
   558→   558→      saveTokenToLocalStorage({
   559→   559→        id: res.token.id,
   560→   560→        name: res.token.name,
   561→   561→        prefix: res.token.prefix,
   562→   562→        secret: res.secret,
   563→   563→        createdAt: res.token.createdAt,
   564→   564→      })
   565→   565→      setCreated(res)
   566→   566→      queryClient.invalidateQueries({ queryKey: ['tokens'] })
   567→   567→    },
   568→   568→    onError: (err) => toastApiError(err, 'Could not create token'),
   569→   569→  })
   570→   570→
   571→   571→  const handleClose = (next: boolean) => {
   572→   572→    if (!next) {
   573→   573→      setName('')
   574→   574→      setCreated(null)
   575→   575→      createMutation.reset()
   576→   576→    }
   577→   577→    onOpenChange(next)
   578→   578→  }
   579→   579→
   580→   580→  const handleCopy = async () => {
   581→   581→    if (!created) return
   582→   582→    try {
   583→   583→      await navigator.clipboard.writeText(created.secret)
   584→   584→      toast.success('Secret copied to clipboard', {
   585→   585→        description: 'Store it in your app config — it will never be shown again.',
   586→   586→      })
   587→   587→    } catch {
   588→   588→      toast.error('Copy failed', { description: 'Select the secret and copy it manually.' })
   589→   589→    }
   590→   590→  }
   591→   591→
   592→   592→  return (
   593→   593→    <Dialog open={open} onOpenChange={handleClose}>
   594→   594→      <DialogContent className="border-zinc-800 bg-zinc-900 sm:max-w-lg">
   595→   595→        {!created ? (
   596→   596→          <>
   597→   597→            <DialogHeader>
   598→   598→              <DialogTitle className="text-base text-zinc-100">New API token</DialogTitle>
   599→   599→              <DialogDescription className="text-xs text-zinc-500">
   600→   600→                Give the token a name so you can recognize it later. The secret is shown once.
   601→   601→              </DialogDescription>
   602→   602→            </DialogHeader>
   603→   603→            <div className="grid gap-2 py-1">
   604→   604→              <Label htmlFor="token-name" className="text-xs text-zinc-400">
   605→   605→                Token name <span className="text-zinc-600">(1–64 characters)</span>
   606→   606→              </Label>
   607→   607→              <Input
   608→   608→                id="token-name"
   609→   609→                value={name}
   610→   610→                onChange={(e) => setName(e.target.value)}
   611→   611→                placeholder="e.g. mobile-app-read"
   612→   612→                maxLength={64}
   613→   613→                autoFocus
   614→   614→                className="border-zinc-700 bg-zinc-950/60 font-mono text-sm"
   615→   615→                onKeyDown={(e) => {
   616→   616→                  if (e.key === 'Enter' && nameValid && !createMutation.isPending) {
   617→   617→                    createMutation.mutate()
   618→   618→                  }
   619→   619→                }}
   620→   620→              />
   621→   621→            </div>
   622→   622→            <DialogFooter>
   623→   623→              <Button
   624→   624→                variant="outline"
   625→   625→                onClick={() => handleClose(false)}
   626→   626→                className="border-zinc-700 bg-zinc-900/60 text-zinc-300 hover:bg-zinc-800"
   627→   627→              >
   628→   628→                Cancel
   629→   629→              </Button>
   630→   630→              <Button
   631→   631→                onClick={() => createMutation.mutate()}
   632→   632→                disabled={!nameValid || createMutation.isPending}
   633→   633→                className="bg-emerald-500 text-zinc-950 hover:bg-emerald-400"
   634→   634→              >
   635→   635→                {createMutation.isPending ? (
   636→   636→                  <Loader2 className="h-4 w-4 animate-spin" aria-hidden />
   637→   637→                ) : (
   638→   638→                  <KeyRound className="h-4 w-4" aria-hidden />
   639→   639→                )}
   640→   640→                Create token
   641→   641→              </Button>
   642→   642→            </DialogFooter>
   643→   643→          </>
   644→   644→        ) : (
   645→   645→          <>
   646→   646→            <DialogHeader>
   647→   647→              <DialogTitle className="flex items-center gap-2 text-base text-zinc-100">
   648→   648→                <KeyRound className="h-4 w-4 text-emerald-400" aria-hidden />
   649→   649→                Token “{created.token.name}” created
   650→   650→              </DialogTitle>
   651→   651→              <DialogDescription className="text-xs text-zinc-500">
   652→   652→                Prefix <span className="font-mono text-zinc-400">{created.token.prefix}…</span> ·
   653→   653→                created {formatRelative(created.token.createdAt)}
   654→   654→              </DialogDescription>
   655→   655→            </DialogHeader>

   682→   682→            <DialogFooter>

   688→   688→              </Button>
   689→   689→            </DialogFooter>
   690→   690→          </>

   692→   692→      </DialogContent>
   693→   693→    </Dialog>
