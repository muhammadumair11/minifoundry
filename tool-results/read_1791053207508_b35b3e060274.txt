   170→   170→function NavLink({
   171→   171→  item,
   172→   172→  rail,
   173→   173→  onNavigate,
   174→   174→}: {
   175→   175→  item: NavItem
   176→   176→  rail: boolean
   177→   177→  onNavigate?: () => void
   178→   178→}) {
   179→   179→  const pathname = usePathname()
   180→   180→  const active = isActive(pathname, item.href)
   181→   181→  const link = (
   182→   182→    <Link
   183→   183→      href={item.href}
   184→   184→      onClick={onNavigate}
   185→   185→      aria-current={active ? 'page' : undefined}
   186→   186→      aria-label={rail ? item.label : undefined}
   187→   187→      className={cn(
   188→   188→        'group relative flex h-8 shrink-0 items-center rounded-md text-[13px] font-medium transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/50',
   189→   189→        rail ? 'w-full justify-center px-0' : 'gap-2.5 px-2.5',
   190→   190→        active
   191→   191→          ? 'bg-accent text-accent-foreground'
   192→   192→          : 'text-muted-foreground hover:bg-accent/60 hover:text-foreground',
   193→   193→      )}
   194→   194→    >
   195→   195→      {active && !rail && (
   196→   196→        <span
   197→   197→          className="absolute inset-y-[7px] left-0 w-[2.5px] rounded-full bg-foreground/70"
   198→   198→          aria-hidden
   199→   199→        />
   200→   200→      )}
   201→   201→      <item.icon
   202→   202→        className={cn(
   203→   203→          'size-4 shrink-0 transition-colors',
   204→   204→          active
   205→   205→            ? 'text-foreground'
   206→   206→            : 'text-muted-foreground group-hover:text-foreground',
   207→   207→        )}
   208→   208→        aria-hidden
   209→   209→      />
   210→   210→      {!rail && <span className="truncate">{item.label}</span>}
   211→   211→    </Link>
   212→   212→  )
   213→   213→
   214→   214→  // Icon-only rail gets a hover tooltip; full sidebar labels speak for themselves.
   215→   215→  if (!rail) return link
   216→   216→  return (
   217→   217→    <Tooltip>
   218→   218→      <TooltipTrigger asChild>{link}</TooltipTrigger>
   219→   219→      <TooltipContent
   220→   220→        side="right"
   221→   221→        sideOffset={10}
   222→   222→        className="border-border bg-popover px-2 py-1 text-xs text-popover-foreground shadow-md"
   223→   223→      >
   224→   224→        {item.label}
   225→   225→      </TooltipContent>
   226→   226→    </Tooltip>
   227→   227→  )
   228→   228→}

   410→   410→/**
   411→   411→ * Anonymous demo mode — "Sign in" button + the legacy role simulator,
   412→   412→ * relabeled "Demo role" with a subtle DEMO badge (task 7-c).
   413→   413→ */
   414→   414→function DemoAuthControls({ session }: { session: SessionResponse | undefined }) {
   415→   415→  const queryClient = useQueryClient()
   416→   416→  const role = session?.role
   417→   417→
   418→   418→  const roleMutation = useMutation({
   419→   419→    mutationFn: (next: FoundryRole) =>
   420→   420→      apiSend<{ role: FoundryRole }>('/api/session/role', 'POST', { role: next }),
   421→   421→    onSuccess: (res) => {
   422→   422→      toast.success(`Now acting as ${res.role}`)
   423→   423→      // Instant feedback, then refetch EVERYTHING so masked data re-renders.
   424→   424→      queryClient.setQueryData<SessionResponse>(['session'], (old) => ({
   425→   425→        role: res.role,
   426→   426→        session: old?.session ?? null,
   427→   427→      }))
   428→   428→      queryClient.invalidateQueries()
   429→   429→    },
   430→   430→    onError: (err) => toastApiError(err, 'Could not switch role'),
   431→   431→  })
   432→   432→
   433→   433→  if (!role) {
   434→   434→    return <Skeleton className="h-7 w-[86px] rounded-md" aria-hidden />
   435→   435→  }
   436→   436→
   437→   437→  const meta = ROLE_META[role]
   438→   438→  return (
   439→   439→    <>
   440→   440→      <Button
   441→   441→        asChild
   442→   442→        variant="outline"
   443→   443→        size="sm"
   444→   444→        className="h-7 gap-1.5 rounded-md px-2 text-xs font-medium"
   445→   445→      >
   446→   446→        <Link href="/login" aria-label="Sign in">
   447→   447→          <LogIn className="size-3.5 shrink-0" aria-hidden />
   448→   448→          <span className="hidden 2xl:inline">Sign in</span>
   449→   449→        </Link>
   450→   450→      </Button>
   451→   451→      <span
   452→   452→        className="hidden rounded border border-dashed border-border px-1 py-0 font-mono text-[9px] leading-4 text-muted-foreground xl:inline"
   453→   453→        aria-label="Demo mode — simulator role"
   454→   454→      >
   455→   455→        DEMO
   456→   456→      </span>
   457→   457→      <Select
   458→   458→        value={role}
   459→   459→        onValueChange={(next) => {
   460→   460→          if (next !== role) roleMutation.mutate(next as FoundryRole)
   461→   461→        }}
   462→   462→      >
   463→   463→        <SelectTrigger
   464→   464→          size="sm"
   465→   465→          aria-label={`Demo role — acting as ${meta.label}, switch`}
   466→   466→          title="Demo role (simulator) — sign in to use a real identity"
   467→   467→          className="h-7 gap-1.5 rounded-md border-border bg-muted/60 px-1.5 text-xs font-medium text-foreground hover:bg-accent data-[size=sm]:h-7"
   468→   468→        >
   469→   469→          <span className={cn('h-1.5 w-1.5 shrink-0 rounded-full', meta.dot)} aria-hidden />
   470→   470→          <span className={cn('hidden leading-none md:inline', meta.text)}>{meta.label}</span>
   471→   471→        </SelectTrigger>
   472→   472→        <SelectContent className="min-w-0">
   473→   473→          {ROLES.map((r) => {

   651→   651→const SIDEBAR_KEY = 'mf.sidebar.collapsed'
   652→   652→
   653→   653→/**
   654→   654→ * Rail/expanded preference, persisted in localStorage. Backed by
   655→   655→ * useSyncExternalStore so the client picks up the stored value during
   656→   656→ * hydration (no effect, no flash) and cross-tab updates stay in sync.
   657→   657→ */
   658→   658→function useSidebarCollapsed(): [boolean, (next: boolean) => void] {
   659→   659→  const subscribe = useCallback((onStoreChange: () => void) => {
   660→   660→    window.addEventListener('mf:sidebar', onStoreChange)
   661→   661→    window.addEventListener('storage', onStoreChange)
   662→   662→    return () => {
   663→   663→      window.removeEventListener('mf:sidebar', onStoreChange)
   664→   664→      window.removeEventListener('storage', onStoreChange)
   665→   665→    }
   666→   666→  }, [])
   667→   667→  const collapsed = useSyncExternalStore(
   668→   668→    subscribe,
   669→   669→    () => window.localStorage.getItem(SIDEBAR_KEY) === '1',
   670→   670→    () => false, // server snapshot: expanded
   671→   671→  )
   672→   672→  const setCollapsed = useCallback((next: boolean) => {
   673→   673→    try {
   674→   674→      window.localStorage.setItem(SIDEBAR_KEY, next ? '1' : '0')
   675→   675→    } catch {
   676→   676→      /* private mode etc. — preference just won't persist */
   677→   677→    }
   678→   678→    window.dispatchEvent(new Event('mf:sidebar'))
   679→   679→  }, [])
   680→   680→  return [collapsed, setCollapsed]
   681→   681→}
