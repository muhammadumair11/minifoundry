   158→   158→function NavLink({
   159→   159→  item,
   160→   160→  rail,
   161→   161→  onNavigate,
   162→   162→}: {
   163→   163→  item: NavItem
   164→   164→  rail: boolean
   165→   165→  onNavigate?: () => void
   166→   166→}) {
   167→   167→  const pathname = usePathname()
   168→   168→  const active = isActive(pathname, item.href)
   169→   169→  const link = (
   170→   170→    <Link
   171→   171→      href={item.href}
   172→   172→      onClick={onNavigate}
   173→   173→      aria-current={active ? 'page' : undefined}
   174→   174→      aria-label={rail ? item.label : undefined}
   175→   175→      className={cn(
   176→   176→        'group relative flex h-8 shrink-0 items-center rounded-md text-[13px] font-medium transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/50',
   177→   177→        rail ? 'w-full justify-center px-0' : 'gap-2.5 px-2.5',
   178→   178→        active
   179→   179→          ? 'bg-emerald-500/[0.12] text-emerald-300'
   180→   180→          : 'text-zinc-400 hover:bg-zinc-900 hover:text-zinc-100',
   181→   181→      )}
   182→   182→    >
   183→   183→      {active && !rail && (
   184→   184→        <span
   185→   185→          className="absolute inset-y-[7px] left-0 w-[2.5px] rounded-full bg-emerald-400"
   186→   186→          aria-hidden
   187→   187→        />
   188→   188→      )}
   189→   189→      <item.icon
   190→   190→        className={cn(
   191→   191→          'size-4 shrink-0 transition-colors',
   192→   192→          active
   193→   193→            ? 'text-emerald-400'
   194→   194→            : 'text-zinc-500 group-hover:text-zinc-300',
   195→   195→        )}
   196→   196→        aria-hidden
   197→   197→      />
   198→   198→      {!rail && <span className="truncate">{item.label}</span>}
   199→   199→    </Link>
   200→   200→  )
   201→   201→
   202→   202→  // Icon-only rail gets a hover tooltip; full sidebar labels speak for themselves.
   203→   203→  if (!rail) return link
   204→   204→  return (
   205→   205→    <Tooltip>
   206→   206→      <TooltipTrigger asChild>{link}</TooltipTrigger>
   207→   207→      <TooltipContent
   208→   208→        side="right"
   209→   209→        sideOffset={10}
   210→   210→        className="border-zinc-800 bg-zinc-900 px-2 py-1 text-xs text-zinc-200"
   211→   211→      >
   212→   212→        {item.label}
   213→   213→      </TooltipContent>
   214→   214→    </Tooltip>
   215→   215→  )
   216→   216→}

   399→   399→ * Anonymous demo mode — "Sign in" button + the legacy role simulator,
   400→   400→ * relabeled "Demo role" with a subtle DEMO badge (task 7-c).
   401→   401→ */
   402→   402→function DemoAuthControls({ session }: { session: SessionResponse | undefined }) {
   403→   403→  const queryClient = useQueryClient()
   404→   404→  const role = session?.role
   405→   405→
   406→   406→  const roleMutation = useMutation({
   407→   407→    mutationFn: (next: FoundryRole) =>
   408→   408→      apiSend<{ role: FoundryRole }>('/api/session/role', 'POST', { role: next }),
   409→   409→    onSuccess: (res) => {
   410→   410→      toast.success(`Now acting as ${res.role}`)
   411→   411→      // Instant feedback, then refetch EVERYTHING so masked data re-renders.
   412→   412→      queryClient.setQueryData<SessionResponse>(['session'], (old) => ({
   413→   413→        role: res.role,
   414→   414→        session: old?.session ?? null,
   415→   415→      }))
   416→   416→      queryClient.invalidateQueries()
   417→   417→    },
   418→   418→    onError: (err) => toastApiError(err, 'Could not switch role'),
   419→   419→  })
   420→   420→
   421→   421→  if (!role) {
   422→   422→    return <Skeleton className="h-7 w-[86px] rounded-md bg-zinc-800" aria-hidden />
   423→   423→  }
   424→   424→
   425→   425→  const meta = ROLE_META[role]
   426→   426→  return (
   427→   427→    <>
   428→   428→      <Button
   429→   429→        asChild
   430→   430→        variant="outline"
   431→   431→        size="sm"
   432→   432→        className="h-7 gap-1.5 rounded-md border-emerald-500/40 bg-emerald-500/10 px-2 text-xs font-medium text-emerald-400 hover:bg-emerald-500/20 hover:text-emerald-300"
   433→   433→      >
   434→   434→        <Link href="/login" aria-label="Sign in">
   435→   435→          <LogIn className="size-3.5 shrink-0" aria-hidden />
   436→   436→          <span className="hidden 2xl:inline">Sign in</span>
   437→   437→        </Link>
   438→   438→      </Button>
   439→   439→      <span
   440→   440→        className="hidden rounded border border-dashed border-zinc-700 px-1 py-0 font-mono text-[9px] leading-4 text-zinc-500 xl:inline"
   441→   441→        aria-label="Demo mode — simulator role"
   442→   442→      >
   443→   443→        DEMO
   444→   444→      </span>
   445→   445→      <Select
   446→   446→        value={role}
   447→   447→        onValueChange={(next) => {
   448→   448→          if (next !== role) roleMutation.mutate(next as FoundryRole)
   449→   449→        }}
   450→   450→      >
   451→   451→        <SelectTrigger
   452→   452→          size="sm"
   453→   453→          aria-label={`Demo role — acting as ${meta.label}, switch`}
   454→   454→          title="Demo role (simulator) — sign in to use a real identity"
   455→   455→          className="h-7 gap-1.5 rounded-md border-zinc-800 bg-zinc-900/60 px-1.5 text-xs font-medium text-zinc-200 hover:bg-zinc-800 data-[size=sm]:h-7"
   456→   456→        >
   457→   457→          <span className={cn('h-1.5 w-1.5 shrink-0 rounded-full', meta.dot)} aria-hidden />
   458→   458→          <span className={cn('hidden leading-none md:inline', meta.text)}>{meta.label}</span>
   459→   459→        </SelectTrigger>
   460→   460→        <SelectContent className="min-w-0 border-zinc-800 bg-zinc-900">
   461→   461→          {ROLES.map((r) => {
   462→   462→            const m = ROLE_META[r]
   463→   463→            const Icon = m.icon
   464→   464→            return (
   465→   465→              <SelectItem
   466→   466→                key={r}
   467→   467→                value={r}

   639→   639→const SIDEBAR_KEY = 'mf.sidebar.collapsed'
   640→   640→
   641→   641→/**
   642→   642→ * Rail/expanded preference, persisted in localStorage. Backed by
   643→   643→ * useSyncExternalStore so the client picks up the stored value during
   644→   644→ * hydration (no effect, no flash) and cross-tab updates stay in sync.
   645→   645→ */
   646→   646→function useSidebarCollapsed(): [boolean, (next: boolean) => void] {
   647→   647→  const subscribe = useCallback((onStoreChange: () => void) => {
   648→   648→    window.addEventListener('mf:sidebar', onStoreChange)
   649→   649→    window.addEventListener('storage', onStoreChange)
   650→   650→    return () => {
   651→   651→      window.removeEventListener('mf:sidebar', onStoreChange)
   652→   652→      window.removeEventListener('storage', onStoreChange)
   653→   653→    }
   654→   654→  }, [])
   655→   655→  const collapsed = useSyncExternalStore(
   656→   656→    subscribe,
   657→   657→    () => window.localStorage.getItem(SIDEBAR_KEY) === '1',
   658→   658→    () => false, // server snapshot: expanded
   659→   659→  )
   660→   660→  const setCollapsed = useCallback((next: boolean) => {
   661→   661→    try {
   662→   662→      window.localStorage.setItem(SIDEBAR_KEY, next ? '1' : '0')
   663→   663→    } catch {
   664→   664→      /* private mode etc. — preference just won't persist */
   665→   665→    }
   666→   666→    window.dispatchEvent(new Event('mf:sidebar'))
   667→   667→  }, [])
   668→   668→  return [collapsed, setCollapsed]
   669→   669→}
