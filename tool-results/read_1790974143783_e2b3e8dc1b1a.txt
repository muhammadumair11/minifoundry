   393→// ── API usage mini-dashboard (Task 7-c) ──────────────────────────────
   394→// Aggregates the token list client-side: total requests, active count,
   395→// most-used active token, per-token emerald bars (width ∝ requestCount vs
   396→// max) and a dimmed "retired" group for revoked tokens.
   397→
   398→function ApiUsageCard({ tokens, loading }: { tokens: ApiTokenInfo[]; loading: boolean }) {
   399→  const active = tokens
   400→    .filter((t) => !t.revokedAt)
   401→    .sort((a, b) => b.requestCount - a.requestCount)
   402→  const retired = tokens.filter((t) => t.revokedAt)
   403→  const totalRequests = tokens.reduce((n, t) => n + t.requestCount, 0)
   404→  const maxRequests = active.length > 0 ? Math.max(...active.map((t) => t.requestCount)) : 0
   405→  const top = active.find((t) => t.requestCount > 0)
   406→  const hasTraffic = totalRequests > 0
   407→
   408→  return (
   409→    <Card className="mb-6 border-zinc-800 bg-zinc-900/60">
   410→      <CardHeader className="gap-1.5 border-b border-zinc-800 p-4 sm:p-5">
   411→        <CardTitle className="flex items-center gap-2 text-sm font-semibold text-zinc-200">
   412→          <Activity className="h-4 w-4 text-emerald-400" aria-hidden />
   413→          API usage
   414→        </CardTitle>
   415→        <CardDescription className="mt-1 text-xs text-zinc-500">
   416→          Live traffic across all tokens — every authenticated{' '}
   417→          <span className="font-mono text-zinc-400">/api/foundry/v1</span> call counts here.
   418→        </CardDescription>
   419→      </CardHeader>
   420→      <CardContent className="p-4 sm:p-5">
   421→        {loading ? (
   422→          <div className="space-y-3">
   423→            <div className="grid grid-cols-1 gap-3 sm:grid-cols-3">
   424→              {[0, 1, 2].map((i) => (
   425→                <Skeleton key={i} className="h-10 w-full bg-zinc-800/70" />
   426→              ))}
   427→            </div>
   428→            <Skeleton className="h-16 w-full bg-zinc-800/70" />
   429→          </div>
   430→        ) : tokens.length === 0 ? (
   431→          <p className="py-2 text-center text-sm text-zinc-500">
   432→            No tokens yet — create one below to start calling the read API.
   433→          </p>
   434→        ) : !hasTraffic ? (
   435→          <div className="flex flex-col items-center gap-1.5 py-4 text-center">
   436→            <Activity className="h-5 w-5 text-zinc-600" aria-hidden />
   437→            <p className="text-sm font-medium text-zinc-400">
   438→              No API traffic yet — try the{' '}
   439→              <Link
   440→                href="/api-docs"
   441→                className="text-emerald-500/80 underline-offset-2 hover:text-emerald-400 hover:underline"
   442→              >
   443→                Explorer
   444→              </Link>
   445→              .
   446→            </p>
   447→            <p className="text-xs text-zinc-600">
   448→              Requests land here per token as soon as apps start calling.
   449→            </p>
   450→          </div>
   451→        ) : (
   452→          <>
   453→            {/* Aggregate stats */}
   454→            <div className="grid grid-cols-1 gap-3 sm:grid-cols-3">
   455→              <div className="rounded-md border border-zinc-800 bg-zinc-950/60 px-3 py-2.5">
   456→                <p className="text-[10px] font-medium uppercase tracking-wider text-zinc-500">
   457→                  Total requests
   458→                </p>
   459→                <p className="mt-0.5 font-mono text-xl font-semibold tabular-nums text-zinc-100">
   460→                  {totalRequests.toLocaleString()}
   461→                </p>
   462→              </div>
   463→              <div className="rounded-md border border-zinc-800 bg-zinc-950/60 px-3 py-2.5">
   464→                <p className="text-[10px] font-medium uppercase tracking-wider text-zinc-500">
   465→                  Active tokens
   466→                </p>
   467→                <p className="mt-0.5 font-mono text-xl font-semibold tabular-nums text-zinc-100">
   468→                  {active.length}
   469→                  <span className="ml-1.5 text-xs font-normal text-zinc-500">
   470→                    of {tokens.length}
   471→                  </span>
   472→                </p>
   473→              </div>
   474→              <div className="min-w-0 rounded-md border border-zinc-800 bg-zinc-950/60 px-3 py-2.5">
   475→                <p className="text-[10px] font-medium uppercase tracking-wider text-zinc-500">
   476→                  Most used
   477→                </p>
   478→                <p className="mt-0.5 truncate font-mono text-sm font-semibold text-emerald-400">
   479→                  {top ? top.name : '—'}
   480→                </p>
   481→                {top && (
   482→                  <p className="font-mono text-[11px] text-zinc-500">
   483→                    {top.requestCount.toLocaleString()} requests
   484→                  </p>
   485→                )}
   486→              </div>
   487→            </div>
   488→
   489→            {/* Per-token bars (active only) */}
   490→            <ul className="mt-4 space-y-2.5">
   491→              {active.map((t) => {
   492→                const pct = maxRequests > 0 ? Math.max((t.requestCount / maxRequests) * 100, t.requestCount > 0 ? 2 : 0) : 0
   493→                return (
   494→                  <li key={t.id} className="flex items-center gap-3">
   495→                    <span
   496→                      className="w-36 shrink-0 truncate text-xs text-zinc-300 sm:w-44"
   497→                      title={t.name}
   498→                    >
   499→                      {t.name}
   500→                    </span>
   501→                    <span className="h-2 min-w-0 flex-1 overflow-hidden rounded-full bg-zinc-800">
   502→                      <span
   503→                        className="block h-full rounded-full bg-emerald-500/80"
   504→                        style={{ width: `${pct}%` }}
   505→                        aria-hidden
   506→                      />
   507→                    </span>
   508→                    <span className="w-14 shrink-0 text-right font-mono text-xs tabular-nums text-zinc-300">
   509→                      {t.requestCount.toLocaleString()}
   510→                    </span>
   511→                  </li>
   512→                )
   513→              })}
   514→            </ul>
   515→
   516→            {/* Retired tokens — dimmed, no bars */}
   517→            {retired.length > 0 && (
   518→              <div className="mt-4 border-t border-zinc-800/70 pt-3">
   519→                <p className="text-[10px] font-medium uppercase tracking-wider text-zinc-600">
   520→                  Retired · {retired.length} revoked
   521→                </p>
   522→                <p className="mt-1 flex flex-wrap gap-x-4 gap-y-0.5 font-mono text-[11px] text-zinc-600">
   523→                  {retired.map((t) => (
   524→                    <span key={t.id}>
   525→                      {t.name} · {t.requestCount.toLocaleString()} req
   526→                    </span>
   527→                  ))}
   528→                </p>
   529→              </div>
   530→            )}
   531→          </>
   532→        )}
   533→      </CardContent>
   534→    </Card>
   535→  )
   536→}

   538→// ── Create dialog (name → POST → show-once secret panel) ────────────
   539→
   540→function CreateTokenDialog({
   541→  open,
   542→  onOpenChange,
   543→}: {
   544→  open: boolean
   545→  onOpenChange: (open: boolean) => void
   546→}) {
   547→  const queryClient = useQueryClient()
   548→  const [name, setName] = useState('')
   549→  const [created, setCreated] = useState<CreateTokenResponse | null>(null)
   550→
   551→  const trimmed = name.trim()
   552→  const nameValid = trimmed.length >= 1 && trimmed.length <= 64
   553→
   554→  const createMutation = useMutation({
   555→    mutationFn: () => apiSend<CreateTokenResponse>('/api/tokens', 'POST', { name: trimmed }),
   556→    onSuccess: (res) => {
   557→      // Remember locally for the Explorer before anything else can unmount.
   558→      saveTokenToLocalStorage({
   559→        id: res.token.id,
   560→        name: res.token.name,
   561→        prefix: res.token.prefix,
   562→        secret: res.secret,
   563→        createdAt: res.token.createdAt,
   564→      })
   565→      setCreated(res)
   566→      queryClient.invalidateQueries({ queryKey: ['tokens'] })
   567→    },
   568→    onError: (err) => toastApiError(err, 'Could not create token'),
   569→  })
   570→
   571→  const handleClose = (next: boolean) => {
   572→    if (!next) {
   573→      setName('')
   574→      setCreated(null)
   575→      createMutation.reset()
   576→    }
   577→    onOpenChange(next)
   578→  }
   579→
   580→  const handleCopy = async () => {
   581→    if (!created) return
   582→    try {
   583→      await navigator.clipboard.writeText(created.secret)
   584→      toast.success('Secret copied to clipboard', {
   585→        description: 'Store it in your app config — it will never be shown again.',
   586→      })
   587→    } catch {
   588→      toast.error('Copy failed', { description: 'Select the secret and copy it manually.' })
   589→    }
   590→  }
   591→
   592→  return (
   593→    <Dialog open={open} onOpenChange={handleClose}>
   594→      <DialogContent className="border-zinc-800 bg-zinc-900 sm:max-w-lg">
   595→        {!created ? (
   596→          <>
   597→            <DialogHeader>
   598→              <DialogTitle className="text-base text-zinc-100">New API token</DialogTitle>
   599→              <DialogDescription className="text-xs text-zinc-500">
   600→                Give the token a name so you can recognize it later. The secret is shown once.
   601→              </DialogDescription>
   602→            </DialogHeader>
   603→            <div className="grid gap-2 py-1">
   604→              <Label htmlFor="token-name" className="text-xs text-zinc-400">
   605→                Token name <span className="text-zinc-600">(1–64 characters)</span>
   606→              </Label>
   607→              <Input
   608→                id="token-name"
   609→                value={name}
   610→                onChange={(e) => setName(e.target.value)}
   611→                placeholder="e.g. mobile-app-read"
   612→                maxLength={64}
   613→                autoFocus
   614→                className="border-zinc-700 bg-zinc-950/60 font-mono text-sm"
   615→                onKeyDown={(e) => {
   616→                  if (e.key === 'Enter' && nameValid && !createMutation.isPending) {
   617→                    createMutation.mutate()
   618→                  }
   619→                }}
   620→              />
   621→            </div>
   622→            <DialogFooter>
   623→              <Button
   624→                variant="outline"
   625→                onClick={() => handleClose(false)}
   626→                className="border-zinc-700 bg-zinc-900/60 text-zinc-300 hover:bg-zinc-800"
   627→              >
   628→                Cancel
   629→              </Button>
   630→              <Button
   631→                onClick={() => createMutation.mutate()}
   632→                disabled={!nameValid || createMutation.isPending}
   633→                className="bg-emerald-500 text-zinc-950 hover:bg-emerald-400"
   634→              >
   635→                {createMutation.isPending ? (
   636→                  <Loader2 className="h-4 w-4 animate-spin" aria-hidden />
   637→                ) : (
   638→                  <KeyRound className="h-4 w-4" aria-hidden />
   639→                )}
   640→                Create token
   641→              </Button>
   642→            </DialogFooter>
   643→          </>
   644→        ) : (
   645→          <>
   646→            <DialogHeader>
   647→              <DialogTitle className="flex items-center gap-2 text-base text-zinc-100">
   648→                <KeyRound className="h-4 w-4 text-emerald-400" aria-hidden />
   649→                Token “{created.token.name}” created
   650→              </DialogTitle>
   651→              <DialogDescription className="text-xs text-zinc-500">
   652→                Prefix <span className="font-mono text-zinc-400">{created.token.prefix}…</span> ·
   653→                created {formatRelative(created.token.createdAt)}
   654→              </DialogDescription>
   655→            </DialogHeader>
   656→
   657→            <div className="rounded-lg border border-emerald-500/40 bg-emerald-500/5 p-4">
   658→              <p className="flex items-start gap-2 text-xs font-medium text-emerald-300">
   659→                <ShieldAlert className="mt-0.5 h-3.5 w-3.5 shrink-0" aria-hidden />
   660→                Copy this secret now — it will never be shown again
   661→              </p>
   662→              <div className="mt-3 flex items-stretch gap-2">
   663→                <code
   664→                  className="foundry-scroll min-w-0 flex-1 overflow-x-auto rounded-md border border-zinc-800 bg-zinc-950 px-3 py-2.5 font-mono text-xs text-emerald-300"
   665→                  aria-label="API token secret"
   666→                >
   667→                  {created.secret}
   668→                </code>
   669→                <Button
   670→                  variant="outline"
   671→                  size="sm"
   672→                  onClick={handleCopy}
   673→                  className="shrink-0 border-emerald-500/40 bg-emerald-500/10 text-emerald-300 hover:bg-emerald-500/20 hover:text-emerald-200"
   674→                  aria-label="Copy secret to clipboard"
   675→                >
   676→                  <Copy className="h-3.5 w-3.5" aria-hidden />
   677→                  Copy
   678→                </Button>
   679→              </div>
   680→            </div>
   681→
   682→            <DialogFooter>
   683→              <Button
   684→                onClick={() => handleClose(false)}
   685→                className="bg-emerald-500 text-zinc-950 hover:bg-emerald-400"
   686→              >
   687→                Done
   688→              </Button>
   689→            </DialogFooter>
   690→          </>
   691→        )}
   692→      </DialogContent>
   693→    </Dialog>
   694→  )
   695→}
