   387→// ── Editor (remounts per workbook via key) ───────────────────────────

   389→function WorkbookEditor({
   390→  transform,
   391→  detail,
   392→  detailLoading,
   393→  detailError,
   394→  datasets,
   395→  canEdit,
   396→  onDeleted,
   397→}: {
   398→  transform: TransformSummary
   399→  detail: TransformDetailResponse | null
   400→  detailLoading: boolean
   401→  detailError: string | null
   402→  datasets: DatasetSummary[]
   403→  canEdit: boolean
   404→  onDeleted: () => void
   405→}) {
   406→  const queryClient = useQueryClient()
   407→  const [draftCode, setDraftCode] = useState(transform.sourceCode)
   408→  const [renameOpen, setRenameOpen] = useState(false)
   409→  const [renameValue, setRenameValue] = useState(transform.name)
   410→  const [deleteOpen, setDeleteOpen] = useState(false)
   411→  const [activeTab, setActiveTab] = useState<'preview' | 'logs' | 'history'>('preview')
   412→  const [result, setResult] = useState<TransformRunResponse | null>(null)
   413→  const textareaRef = useRef<HTMLTextAreaElement>(null)
   414→
   415→  const dirty = draftCode !== transform.sourceCode
   416→
   417→  const runsQuery = useQuery<TransformRunsListResponse>({
   418→    queryKey: ['transform-runs', transform.id],
   419→    queryFn: () =>
   420→      apiJson<TransformRunsListResponse>(`/api/transforms/${transform.id}/runs?limit=20`),
   421→  })
   422→
   423→  // The run shown in Preview/Logs: the fresh in-memory run, else the latest
   424→  // run from the detail payload.
   425→  const displayedRun: TransformRunSummary | null =
   426→    result?.run ?? detail?.runs[0] ?? null
   427→  const hasPreviewRows = !!result && result.preview.length >= 0 && result.run.status === 'success'
   428→
   429→  // Output dataset chip — resolve the id from the latest successful run and
   430→  // the display name from the datasets cache.
   431→  const outputDatasetId =
   432→    (result?.run.status === 'success' ? result.run.outputDatasetId : null) ??
   433→    detail?.runs.find((r) => r.status === 'success' && r.outputDatasetId)?.outputDatasetId ??
   434→    null
   435→  const outputDataset = datasets.find((d) => d.id === outputDatasetId) ?? null
   436→
   437→  const invalidateAfterChange = (id: string) => {
   438→    queryClient.invalidateQueries({ queryKey: ['transform', id] })
   439→    queryClient.invalidateQueries({ queryKey: ['transforms'] })
   440→    queryClient.invalidateQueries({ queryKey: ['transform-runs', id] })
   441→    queryClient.invalidateQueries({ queryKey: ['datasets'] })
   442→    queryClient.invalidateQueries({ queryKey: ['stats'] })
   443→  }
   444→
   445→  const runMutation = useMutation({
   446→    // Save-then-run (pipeline-builder convention): run always executes the
   447→    // SAVED source code, so flush a dirty draft first.
   448→    mutationFn: async () => {
   449→      if (dirty) {
   450→        await apiSend<TransformDetailResponse>(`/api/transforms/${transform.id}`, 'PATCH', {
   451→          sourceCode: draftCode,
   452→        })
   453→      }
   454→      return apiSend<TransformRunResponse>(`/api/transforms/${transform.id}/run`, 'POST')
   455→    },
   456→    onSuccess: (res) => {
   457→      setResult(res)
   458→      if (res.run.status === 'success') {
   459→        setActiveTab('preview')
   460→        toast.success('Run succeeded', {
   461→          description: `${(res.run.rowCount ?? 0).toLocaleString()} rows → ${transform.outputDatasetName}`,
   462→        })
   463→      } else {
   464→        setActiveTab('logs')
   465→        toast.error('Run failed', {
   466→          description: res.run.error ?? 'Check the Logs tab for details.',
   467→        })
   468→      }
   469→      invalidateAfterChange(transform.id)
   470→    },
   471→    onError: (err) => toastApiError(err, 'Run request failed'),
   472→  })
   473→
   474→  const saveMutation = useMutation({
   475→    mutationFn: () =>
   476→      apiSend<TransformDetailResponse>(`/api/transforms/${transform.id}`, 'PATCH', {
   477→        sourceCode: draftCode,
   478→      }),
   479→    onSuccess: () => {
   480→      toast.success('Workbook saved', { description: transform.name })
   481→      invalidateAfterChange(transform.id)
   482→    },
   483→    onError: (err) => toastApiError(err, 'Save failed'),
   484→  })
   485→
   486→  const renameMutation = useMutation({
   487→    mutationFn: (name: string) =>
   488→      apiSend<TransformDetailResponse>(`/api/transforms/${transform.id}`, 'PATCH', { name }),
   489→    onSuccess: () => {
   490→      toast.success('Workbook renamed', { description: renameValue.trim() })
   491→      setRenameOpen(false)
   492→      invalidateAfterChange(transform.id)
   493→    },
   494→    onError: (err) => toastApiError(err, 'Rename failed'),
   495→  })
   496→
   497→  const deleteMutation = useMutation({
   498→    mutationFn: () => apiSend<{ ok: boolean }>(`/api/transforms/${transform.id}`, 'DELETE'),
   499→    onSuccess: () => {
   500→      toast.success('Workbook deleted', { description: `${transform.name} — runs and history removed` })
   501→      setDeleteOpen(false)
   502→      queryClient.invalidateQueries({ queryKey: ['transforms'] })
   503→      queryClient.invalidateQueries({ queryKey: ['datasets'] })
   504→      onDeleted()
   505→    },
   506→    onError: (err) => toastApiError(err, 'Delete failed'),
   507→  })
   508→
   509→  const run = () => {
   510→    if (!canEdit || runMutation.isPending) return
   511→    runMutation.mutate()
   512→  }
   513→
   514→  const handleEditorKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   515→    if (e.key === 'Tab') {
   516→      e.preventDefault()
   517→      const el = e.currentTarget
   518→      const { selectionStart, selectionEnd, value } = el
   519→      const next = `${value.slice(0, selectionStart)}  ${value.slice(selectionEnd)}`
   520→      setDraftCode(next)
   521→      requestAnimationFrame(() => {
   522→        el.selectionStart = el.selectionEnd = selectionStart + 2
   523→      })
   524→      return
   525→    }
   526→    if ((e.metaKey || e.ctrlKey) && e.key === 'Enter') {
   527→      e.preventDefault()
   528→      run()
   529→    }
   530→  }
   531→
   532→  const submitRename = () => {
   533→    const next = renameValue.trim()
   534→    if (!next || next === transform.name || renameMutation.isPending) return
   535→    renameMutation.mutate(next)
   536→  }
   537→
   538→  const latestRunStatus = transform.latestRun?.status ?? null

   965→function PreviewTable({
   966→  result,
   967→  outputName,
   968→}: {
   969→  result: TransformRunResponse
   970→  outputName: string
   971→}) {
   972→  const columns = result.schema
   973→  const rows = result.preview
   974→  const typeByName = useMemo(() => {
   975→    const m = new Map<string, string>()
   976→    for (const c of columns) m.set(c.name, c.type)
   977→    return m
   978→  }, [columns])
   979→
   980→  const renderCell = (value: unknown, colName: string) => {
   981→    if (value === null || value === undefined || value === '') {
   982→      return <span className="text-zinc-700">∅</span>
   983→    }
   984→    const type = typeByName.get(colName)
   985→    if (typeof value === 'boolean' || type === 'boolean') {
   986→      return (
   987→        <span className={value ? 'text-amber-400' : 'text-zinc-500'}>{String(value)}</span>
   988→      )
   989→    }
   990→    if (typeof value === 'number' || type === 'number') {
   991→      const n = typeof value === 'number' ? value : Number(value)
   992→      return <span className="tabular-nums">{Number.isNaN(n) ? String(value) : n.toLocaleString()}</span>
   993→    }
   994→    return String(value)
   995→  }
   996→
   997→  return (
   998→    <div className="rounded-md border border-zinc-800 bg-zinc-950/50">
   999→      <div className="flex flex-wrap items-center justify-between gap-2 border-b border-zinc-800 px-3 py-2">
  1000→        <p className="flex items-center gap-1.5 font-mono text-[11px] text-zinc-400">
  1001→          <Braces className="h-3.5 w-3.5 text-emerald-400" aria-hidden />
  1002→          output → <span className="text-emerald-400">{outputName}</span>
  1003→          {result.run.outputVersionNumber !== null && (
  1004→            <span className="rounded border border-zinc-800 bg-zinc-900 px-1 text-[10px] text-zinc-500">
  1005→              v{result.run.outputVersionNumber}
  1006→            </span>
  1007→          )}
  1008→        </p>
  1009→        <p className="font-mono text-[10px] text-zinc-500">
  1010→          showing {rows.length} of {(result.run.rowCount ?? rows.length).toLocaleString()} rows ·{' '}
  1011→          {result.run.durationMs !== null ? `${result.run.durationMs}ms` : '—'}
  1012→        </p>
  1013→      </div>
  1014→      <div className="foundry-scroll max-h-80 overflow-auto">
