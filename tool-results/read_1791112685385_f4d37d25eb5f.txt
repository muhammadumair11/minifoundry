   197→/** "api: …" chip — ONE adaptive entry point (merged in 8-d): admins get the
   198→    24h usage window ("N req / 24h · M tokens", → /api-tokens usage dashboard);
   199→    non-admins (or admins before usage loads) get token/Explorer labels
   200→    ("M tokens · v1" / "v1 read endpoints", → /api-docs). Queries are gated on
   201→    the effective role so they don't 403 for viewer/editor. */
   202→function ApiChip() {
   203→  // Same ['session'] pattern as site-header — `role` is the effective role.
   204→  const { data: session } = useQuery<SessionResponse>({
   205→    queryKey: ['session'],
   206→    queryFn: () => apiJson<SessionResponse>('/api/session'),
   207→    staleTime: 30_000,
   208→  })
   209→  const isAdmin = session?.role === 'admin'
   210→  const tokensQuery = useQuery<TokensListResponse>({
   211→    queryKey: ['tokens'],
   212→    queryFn: () => apiJson<TokensListResponse>('/api/tokens'),
   213→    enabled: isAdmin,
   214→    retry: false,
   215→    staleTime: 30_000,
   216→  })
   217→  const usageQuery = useQuery<ApiUsageResponse>({
   218→    queryKey: ['api-usage'],
   219→    queryFn: () => apiJson<ApiUsageResponse>('/api/tokens/usage'),
   220→    enabled: isAdmin,
   221→    retry: false,
   222→  })
   223→  const activeCount = tokensQuery.data?.tokens.filter((t) => !t.revokedAt).length ?? 0
   224→  const total = usageQuery.data?.total
   225→  const href = isAdmin ? '/api-tokens' : '/api-docs'
   226→  const label = isAdmin
   227→    ? total !== undefined
   228→      ? total > 0
   229→        ? `${total.toLocaleString()} req / 24h · ${activeCount} tokens`
   230→        : `idle · ${activeCount} tokens`
   231→      : `${activeCount} tokens · v1`
   232→    : 'v1 read endpoints'
   233→  return (
   234→    <Link
   235→      href={href}
   236→      className="inline-flex items-center gap-1.5 rounded-md border border-border bg-card px-3 py-1.5 font-mono text-[11px] text-muted-foreground transition-colors hover:bg-accent hover:text-foreground"
   237→      aria-label={
   238→        isAdmin
   239→          ? `API usage — ${total ?? 0} requests in the last 24h, ${activeCount} active tokens. Open API tokens.`
   240→          : 'Public read API — open the API Explorer.'
   241→      }
   242→    >
   243→      <Activity className="h-3 w-3 text-muted-foreground" aria-hidden />
   244→      <span>
   245→        api:{' '}
   246→        <span className="text-foreground">{label}</span>
   247→      </span>
   248→    </Link>
   249→  )
   250→}
