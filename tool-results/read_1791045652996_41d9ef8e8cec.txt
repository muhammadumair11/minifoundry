    67→function useDebouncedValue<T>(value: T, delayMs: number): T {
    68→  const [debounced, setDebounced] = useState(value)
    69→  useEffect(() => {
    70→    const t = setTimeout(() => setDebounced(value), delayMs)
    71→    return () => clearTimeout(t)
    72→  }, [value, delayMs])
    73→  return debounced
    74→}
    75→
    76→function renderCellValue(value: unknown, type: string | undefined): React.ReactNode {
    77→  if (value === null || value === undefined || value === '') {
    78→    return <span className="text-zinc-700">∅</span>
    79→  }
    80→  if (value === MASK_LITERAL) {
    81→    return <span className="italic text-zinc-600">{MASK_LITERAL}</span>
    82→  }
    83→  if (typeof value === 'boolean' || type === 'boolean') {
    84→    return (
    85→      <span className={value ? 'text-amber-400' : 'text-zinc-500'}>
    86→        {String(value)}
    87→      </span>
    88→    )
    89→  }
    90→  if (typeof value === 'number') {
    91→    return <span className="tabular-nums">{value.toLocaleString()}</span>
    92→  }
    93→  return String(value)
    94→}
