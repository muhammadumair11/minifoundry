   170→function NavLink({
   171→  item,
   172→  rail,
   173→  onNavigate,
   174→}: {
   175→  item: NavItem
   176→  rail: boolean
   177→  onNavigate?: () => void
   178→}) {
   179→  const pathname = usePathname()
   180→  const active = isActive(pathname, item.href)
   181→  const link = (
   182→    <Link
   183→      href={item.href}
   184→      onClick={onNavigate}
   185→      aria-current={active ? 'page' : undefined}
   186→      aria-label={rail ? item.label : undefined}
   187→      className={cn(
   188→        'group relative flex h-8 shrink-0 items-center rounded-md text-[13px] font-medium transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/50',
   189→        rail ? 'w-full justify-center px-0' : 'gap-2.5 px-2.5',
   190→        active
   191→          ? 'bg-accent text-accent-foreground'
   192→          : 'text-muted-foreground hover:bg-accent/60 hover:text-foreground',
   193→      )}
   194→    >
   195→      {active && !rail && (
   196→        <span
   197→          className="absolute inset-y-[7px] left-0 w-[2.5px] rounded-full bg-foreground/70"
   198→          aria-hidden
   199→        />
   200→      )}
   201→      <item.icon
   202→        className={cn(
   203→          'size-4 shrink-0 transition-colors',
   204→          active
   205→            ? 'text-foreground'
   206→            : 'text-muted-foreground group-hover:text-foreground',
   207→        )}
   208→        aria-hidden
   209→      />
   210→      {!rail && <span className="truncate">{item.label}</span>}
   211→    </Link>
   212→  )
   213→
   214→  // Icon-only rail gets a hover tooltip; full sidebar labels speak for themselves.
   215→  if (!rail) return link
   216→  return (
   217→    <Tooltip>
   218→      <TooltipTrigger asChild>{link}</TooltipTrigger>
   219→      <TooltipContent
   220→        side="right"
   221→        sideOffset={10}
   222→        className="border-border bg-popover px-2 py-1 text-xs text-popover-foreground shadow-md"
   223→      >
   224→        {item.label}
   225→      </TooltipContent>
   226→    </Tooltip>
   227→  )
   228→}

   414→function DemoAuthControls({ session }: { session: SessionResponse | undefined }) {

   418→  const roleMutation = useMutation({

   424→      queryClient.setQueryData<SessionResponse>(['session'], (old) => ({
   425→        role: res.role,
   426→        session: old?.session ?? null,
   427→      }))

   494→/**
   495→ * Signed-in identity — avatar menu (task 7-c). Shows the real identity
   496→ * (name/email/role badge), a "Switch role (admin)" impersonation submenu for
   497→ * session admins, and Sign out (full cache invalidation so role-gated data
   498→ * downgrades back to demo level).
   499→ */
   500→function UserMenu({ session }: { session: SessionResponse }) {
   501→  const queryClient = useQueryClient()
   502→  const user: SessionUserInfo | null = session.session ?? null
   503→  // EFFECTIVE role (impersonation-aware); `user.role` stays the TRUE role.
   504→  const effectiveRole = user?.effectiveRole ?? session.role
   505→  const isAdminUser = user?.role === 'admin'
   506→  const impersonating = Boolean(user?.impersonating)
   507→
   508→  const logoutMutation = useMutation({
   509→    mutationFn: () => apiSend<{ ok: true }>('/api/auth/logout', 'POST'),
   510→    onSuccess: async () => {
   511→      // Demo-safe default: drop the simulator cookie to viewer so the
   512→      // anonymous session starts least-privileged (the bare no-cookie
   513→      // machine default would be admin). Best effort — ignore failures.
   514→      try {
   515→        await apiSend('/api/session/role', 'POST', { role: 'viewer' })
   516→      } catch {
   517→        /* demo default only — the refetch below settles the UI either way */
   518→      }
   519→      toast.success('Signed out', {
   520→        description: 'Demo mode — acting as viewer. Sign in to use a real identity.',
   521→      })
   522→      queryClient.invalidateQueries()
   523→    },
   524→    onError: (err) => toastApiError(err, 'Could not sign out'),
   525→  })
   526→
   527→  // Impersonation is REAL (session-carried): the server re-issues the session
   528→  // token with the effective role, so refetching ['session'] brings the truth
   529→  // back and every role-gated query re-resolves against the new role.
   530→  const switchRoleMutation = useMutation({
   531→    mutationFn: (next: FoundryRole) =>
   532→      apiSend<{ role: FoundryRole }>('/api/session/role', 'POST', { role: next }),
   533→    onSuccess: (res) => {
   534→      toast.success(
   535→        res.role === 'admin' ? 'Restored admin role' : `Acting as ${res.role}`,
   536→        {
   537→          description:
   538→            res.role === 'admin'
   539→              ? 'Full admin permissions are back.'
   540→              : 'API permissions now follow this role until you switch back or sign out.',
   541→        },
   542→      )
   543→      queryClient.invalidateQueries()
   544→    },
   545→    onError: (err) => toastApiError(err, 'Could not switch role'),
   546→  })
   547→
   548→  if (!user) return null
   549→  const initials = user.name
   550→    .split(/\s+/)
   551→    .filter(Boolean)
   552→    .map((w) => w[0])
   553→    .slice(0, 2)
   554→    .join('')
   555→    .toUpperCase()
   556→
   557→  return (
   558→    <DropdownMenu>
   559→      <DropdownMenuTrigger asChild>
   560→        <button
   561→          className="flex h-8 shrink-0 items-center gap-1.5 rounded-full border border-border bg-card py-1 pl-1 pr-1.5 transition-colors hover:bg-accent focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/50"
   562→          aria-label={`Account menu — ${user.name} (${effectiveRole}${impersonating ? ', impersonated' : ''})`}
   563→        >
   564→          <span
   565→            className="flex size-6 items-center justify-center rounded-full bg-primary font-mono text-[9px] font-bold text-primary-foreground"
   566→            aria-hidden
   567→          >
   568→            {initials}

   651→const SIDEBAR_KEY = 'mf.sidebar.collapsed'
   652→
   653→/**
   654→ * Rail/expanded preference, persisted in localStorage. Backed by
   655→ * useSyncExternalStore so the client picks up the stored value during
   656→ * hydration (no effect, no flash) and cross-tab updates stay in sync.
   657→ */
   658→function useSidebarCollapsed(): [boolean, (next: boolean) => void] {
   659→  const subscribe = useCallback((onStoreChange: () => void) => {
   660→    window.addEventListener('mf:sidebar', onStoreChange)
   661→    window.addEventListener('storage', onStoreChange)
   662→    return () => {
   663→      window.removeEventListener('mf:sidebar', onStoreChange)
   664→      window.removeEventListener('storage', onStoreChange)
   665→    }
   666→  }, [])
   667→  const collapsed = useSyncExternalStore(
   668→    subscribe,
   669→    () => window.localStorage.getItem(SIDEBAR_KEY) === '1',
   670→    () => false, // server snapshot: expanded
   671→  )
   672→  const setCollapsed = useCallback((next: boolean) => {
   673→    try {
   674→      window.localStorage.setItem(SIDEBAR_KEY, next ? '1' : '0')
   675→    } catch {
   676→      /* private mode etc. — preference just won't persist */
   677→    }
   678→    window.dispatchEvent(new Event('mf:sidebar'))
   679→  }, [])
   680→  return [collapsed, setCollapsed]
   681→}
