  1121→export function shortId(id: string | null | undefined, len = 8): string {

  1126→export function formatDuration(startedAt: string, finishedAt: string | null): string {
  1127→  if (!finishedAt) return '—'
  1128→  const ms = new Date(finishedAt).getTime() - new Date(startedAt).getTime()
  1129→  if (ms < 1000) return `${ms}ms`
  1130→  if (ms < 60_000) return `${(ms / 1000).toFixed(1)}s`
  1131→  const m = Math.floor(ms / 60_000)
  1132→  const s = Math.round((ms % 60_000) / 1000)
  1133→  return `${m}m ${s}s`
  1134→}
  1135→
  1136→/** Strict relative time ("3m ago") for data-UI timestamps; '—' for null/invalid. */
  1137→export function formatRelative(iso: string | null | undefined): string {
  1138→  if (!iso) return '—'
  1139→  const d = new Date(iso)
  1140→  if (Number.isNaN(d.getTime())) return '—'
  1141→  return formatDistanceToNowStrict(d, { addSuffix: true })
  1142→}
  1143→
  1144→/** "every 10m" / "every 1h" / "every 1d" — human label for schedule intervals. */
  1145→export function formatInterval(minutes: number): string {
  1146→  if (minutes >= 1440 && minutes % 1440 === 0) return `every ${minutes / 1440}d`
  1147→  if (minutes >= 60 && minutes % 60 === 0) return `every ${minutes / 60}h`
  1148→  return `every ${minutes}m`
  1149→}

  1151→// Data-freshness thresholds (Task 2-c spec): fresh < 15m, aging < 2h, else stale.
  1152→export const FRESH_THRESHOLD_MINUTES = 15
  1153→export const STALE_THRESHOLD_MINUTES = 120
  1154→
  1155→export type FreshnessLevel = 'fresh' | 'aging' | 'stale'
  1156→
  1157→export function freshnessLevel(freshnessMinutes: number): FreshnessLevel {
  1158→  if (freshnessMinutes < FRESH_THRESHOLD_MINUTES) return 'fresh'
  1159→  if (freshnessMinutes < STALE_THRESHOLD_MINUTES) return 'aging'
  1160→  return 'stale'
  1161→}
