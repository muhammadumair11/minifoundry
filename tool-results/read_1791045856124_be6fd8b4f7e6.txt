   492→function WorkbookEditor({
   493→  transform,
   494→  detail,
   495→  detailLoading,
   496→  detailError,
   497→  datasets,
   498→  canEdit,
   499→  onDeleted,
   500→}: {
   501→  transform: TransformSummary
   502→  detail: TransformDetailResponse | null
   503→  detailLoading: boolean
   504→  detailError: string | null
   505→  datasets: DatasetSummary[]
   506→  canEdit: boolean
   507→  onDeleted: () => void
   508→}) {
   509→  const queryClient = useQueryClient()
   510→  const [draftCode, setDraftCode] = useState(transform.sourceCode)
   511→  const [renameOpen, setRenameOpen] = useState(false)
   512→  const [renameValue, setRenameValue] = useState(transform.name)
   513→  const [deleteOpen, setDeleteOpen] = useState(false)
   514→  const [activeTab, setActiveTab] = useState<'preview' | 'logs' | 'history'>('preview')
   515→  const [result, setResult] = useState<TransformRunResponse | null>(null)
   516→  const [resultsOpen, setResultsOpen] = useState(true)
   517→  const [, setFocus] = useFocusMode()
   518→
   519→  const dirty = draftCode !== transform.sourceCode
   520→
   521→  // Schema of the input datasets — feeds Monaco's schema-aware SQL
   522→  // IntelliSense (tables named after datasets + real columns).
   523→  const schemaQuery = useQuery<SchemaTable[]>({
   524→    queryKey: ['transform-input-schema', transform.id],
   525→    queryFn: async () => {
   526→      const tables = await Promise.all(
   527→        transform.inputs.map(async (inp): Promise<SchemaTable | null> => {
   528→          if (!inp.datasetId || !inp.datasetName) return null
   529→          try {
   530→            const d = await apiJson<DatasetDetailResponse>(`/api/datasets/${inp.datasetId}`)
   531→            return {
   532→              name: inp.datasetName,
   533→              columns: d.current.schema.map((c) => ({ name: c.name, type: c.type })),
   534→            }
   535→          } catch {
   536→            return null
   537→          }
   538→        }),
   539→      )
   540→      return tables.filter((t): t is SchemaTable => t !== null)
   541→    },

  1130→function PreviewTable({
  1131→  result,
  1132→  outputName,
  1133→}: {
  1134→  result: TransformRunResponse
  1135→  outputName: string
  1136→}) {
  1137→  const columns = result.schema
  1138→  const rows = result.preview
  1139→  const typeByName = useMemo(() => {
  1140→    const m = new Map<string, string>()
  1141→    for (const c of columns) m.set(c.name, c.type)
  1142→    return m
  1143→  }, [columns])
  1144→
  1145→  const renderCell = (value: unknown, colName: string) => {
  1146→    if (value === null || value === undefined || value === '') {
  1147→      return <span className="text-zinc-700">∅</span>
  1148→    }
  1149→    const type = typeByName.get(colName)
  1150→    if (typeof value === 'boolean' || type === 'boolean') {
  1151→      return (
  1152→        <span className={value ? 'text-amber-400' : 'text-zinc-500'}>{String(value)}</span>
  1153→      )
  1154→    }
  1155→    if (typeof value === 'number' || type === 'number') {
  1156→      const n = typeof value === 'number' ? value : Number(value)
  1157→      return <span className="tabular-nums">{Number.isNaN(n) ? String(value) : n.toLocaleString()}</span>
  1158→    }
  1159→    return String(value)
  1160→  }
  1161→
  1162→  return (
  1163→    <div className="rounded-md border border-zinc-800 bg-zinc-950/50">
  1164→      <div className="flex flex-wrap items-center justify-between gap-2 border-b border-zinc-800 px-3 py-2">
  1165→        <p className="flex items-center gap-1.5 font-mono text-[11px] text-zinc-400">
  1166→          <Braces className="h-3.5 w-3.5 text-emerald-400" aria-hidden />
  1167→          output → <span className="text-emerald-400">{outputName}</span>
  1168→          {result.run.outputVersionNumber !== null && (
  1169→            <span className="rounded border border-zinc-800 bg-zinc-900 px-1 text-[10px] text-zinc-500">
  1170→              v{result.run.outputVersionNumber}
  1171→            </span>
  1172→          )}
  1173→        </p>
  1174→        <p className="font-mono text-[10px] text-zinc-500">
  1175→          showing {rows.length} of {(result.run.rowCount ?? rows.length).toLocaleString()} rows ·{' '}
  1176→          {result.run.durationMs !== null ? `${result.run.durationMs}ms` : '—'}
  1177→        </p>
  1178→      </div>
  1179→      <div className="foundry-scroll max-h-80 overflow-auto lg:max-h-none">
  1180→        <Table>
  1181→          <TableHeader className="sticky top-0 z-10 bg-zinc-900">
  1182→            <TableRow className="border-zinc-800 hover:bg-transparent">
  1183→              {columns.map((col) => (
  1184→                <TableHead
  1185→                  key={col.name}
  1186→                  className={cn(
  1187→                    'h-8 whitespace-nowrap px-3 font-mono text-[11px] text-zinc-400',

  1189→                  )}
  1190→                >
  1191→                  {col.name}
  1192→                  <span className="ml-1 text-zinc-600">
  1193→                    {col.type === 'number' ? '#' : col.type === 'boolean' ? '⌥' : 'A'}
  1194→                  </span>
  1195→                </TableHead>

  1197→            </TableRow>

  1212→                ))}
  1213→              </TableRow>
  1214→            ))}
  1215→          </TableBody>
  1216→        </Table>

  1218→    </div>
