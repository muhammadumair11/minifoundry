   407→function WorkbookEditor({
   408→  transform,
   409→  detail,
   410→  detailLoading,
   411→  detailError,
   412→  datasets,
   413→  canEdit,
   414→  onDeleted,
   415→}: {
   416→  transform: TransformSummary
   417→  detail: TransformDetailResponse | null
   418→  detailLoading: boolean
   419→  detailError: string | null
   420→  datasets: DatasetSummary[]
   421→  canEdit: boolean
   422→  onDeleted: () => void
   423→}) {
   424→  const queryClient = useQueryClient()
   425→  const [draftCode, setDraftCode] = useState(transform.sourceCode)
   426→  const [renameOpen, setRenameOpen] = useState(false)
   427→  const [renameValue, setRenameValue] = useState(transform.name)
   428→  const [deleteOpen, setDeleteOpen] = useState(false)
   429→  const [activeTab, setActiveTab] = useState<'preview' | 'logs' | 'history'>('preview')
   430→  const [result, setResult] = useState<TransformRunResponse | null>(null)
   431→  const textareaRef = useRef<HTMLTextAreaElement>(null)
   432→
   433→  const dirty = draftCode !== transform.sourceCode
   434→
   435→  const runsQuery = useQuery<TransformRunsListResponse>({
   436→    queryKey: ['transform-runs', transform.id],
   437→    queryFn: () =>
   438→      apiJson<TransformRunsListResponse>(`/api/transforms/${transform.id}/runs?limit=20`),
   439→  })
   440→
   441→  // The run shown in Preview/Logs: the fresh in-memory run, else the latest
   442→  // run from the detail payload.
   443→  const displayedRun: TransformRunSummary | null =
   444→    result?.run ?? detail?.runs[0] ?? null
   445→  const hasPreviewRows = !!result && result.preview.length >= 0 && result.run.status === 'success'
   446→
   447→  // Output dataset chip — resolve the id from the latest successful run and
   448→  // the display name from the datasets cache.
   449→  const outputDatasetId =
   450→    (result?.run.status === 'success' ? result.run.outputDatasetId : null) ??
   451→    detail?.runs.find((r) => r.status === 'success' && r.outputDatasetId)?.outputDatasetId ??
   452→    null
   453→  const outputDataset = datasets.find((d) => d.id === outputDatasetId) ?? null
   454→
   455→  const invalidateAfterChange = (id: string) => {
   456→    queryClient.invalidateQueries({ queryKey: ['transform', id] })
   457→    queryClient.invalidateQueries({ queryKey: ['transforms'] })
   458→    queryClient.invalidateQueries({ queryKey: ['transform-runs', id] })
   459→    queryClient.invalidateQueries({ queryKey: ['datasets'] })
   460→    queryClient.invalidateQueries({ queryKey: ['stats'] })
   461→  }
   462→
   463→  const runMutation = useMutation({
   464→    // Save-then-run (pipeline-builder convention): run always executes the
   465→    // SAVED source code, so flush a dirty draft first.
   466→    mutationFn: async () => {
   467→      if (dirty) {
   468→        await apiSend<TransformDetailResponse>(`/api/transforms/${transform.id}`, 'PATCH', {
   469→          sourceCode: draftCode,
   470→        })
   471→      }
   472→      return apiSend<TransformRunResponse>(`/api/transforms/${transform.id}/run`, 'POST')
   473→    },
   474→    onSuccess: (res) => {
   475→      setResult(res)
   476→      if (res.run.status === 'success') {
   477→        setActiveTab('preview')
   478→        toast.success('Run succeeded', {
   479→          description: `${(res.run.rowCount ?? 0).toLocaleString()} rows → ${transform.outputDatasetName}`,
   480→        })
   481→      } else {
   482→        setActiveTab('logs')
   483→        toast.error('Run failed', {
   484→          description: res.run.error ?? 'Check the Logs tab for details.',
   485→        })
   486→      }
   487→      invalidateAfterChange(transform.id)
   488→    },
   489→    onError: (err) => toastApiError(err, 'Run request failed'),
   490→  })
   491→
   492→  const saveMutation = useMutation({
   493→    mutationFn: () =>
   494→      apiSend<TransformDetailResponse>(`/api/transforms/${transform.id}`, 'PATCH', {
   495→        sourceCode: draftCode,
   496→      }),
   497→    onSuccess: () => {
   498→      toast.success('Workbook saved', { description: transform.name })
   499→      invalidateAfterChange(transform.id)
   500→    },
   501→    onError: (err) => toastApiError(err, 'Save failed'),
   502→  })
   503→
   504→  const renameMutation = useMutation({
   505→    mutationFn: (name: string) =>
   506→      apiSend<TransformDetailResponse>(`/api/transforms/${transform.id}`, 'PATCH', { name }),
   507→    onSuccess: () => {
   508→      toast.success('Workbook renamed', { description: renameValue.trim() })
   509→      setRenameOpen(false)
   510→      invalidateAfterChange(transform.id)
   511→    },
   512→    onError: (err) => toastApiError(err, 'Rename failed'),
   513→  })
   514→
   515→  const deleteMutation = useMutation({
   516→    mutationFn: () => apiSend<{ ok: boolean }>(`/api/transforms/${transform.id}`, 'DELETE'),
   517→    onSuccess: () => {
   518→      toast.success('Workbook deleted', { description: `${transform.name} — runs and history removed` })
   519→      setDeleteOpen(false)
   520→      queryClient.invalidateQueries({ queryKey: ['transforms'] })
   521→      queryClient.invalidateQueries({ queryKey: ['datasets'] })
   522→      onDeleted()
   523→    },
   524→    onError: (err) => toastApiError(err, 'Delete failed'),
   525→  })
   526→
   527→  const run = () => {
   528→    if (!canEdit || runMutation.isPending) return
   529→    runMutation.mutate()
   530→  }
   531→
   532→  const handleEditorKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   533→    if (e.key === 'Tab') {
   534→      e.preventDefault()
   535→      const el = e.currentTarget
   536→      const { selectionStart, selectionEnd, value } = el

   985→function PreviewTable({
   986→  result,
   987→  outputName,
   988→}: {
   989→  result: TransformRunResponse
   990→  outputName: string
   991→}) {
   992→  const columns = result.schema
   993→  const rows = result.preview
   994→  const typeByName = useMemo(() => {
   995→    const m = new Map<string, string>()
   996→    for (const c of columns) m.set(c.name, c.type)
   997→    return m
   998→  }, [columns])
   999→
  1000→  const renderCell = (value: unknown, colName: string) => {
  1001→    if (value === null || value === undefined || value === '') {
  1002→      return <span className="text-zinc-700">∅</span>
  1003→    }
  1004→    const type = typeByName.get(colName)
  1005→    if (typeof value === 'boolean' || type === 'boolean') {
  1006→      return (
  1007→        <span className={value ? 'text-amber-400' : 'text-zinc-500'}>{String(value)}</span>
  1008→      )
  1009→    }
  1010→    if (typeof value === 'number' || type === 'number') {
  1011→      const n = typeof value === 'number' ? value : Number(value)
  1012→      return <span className="tabular-nums">{Number.isNaN(n) ? String(value) : n.toLocaleString()}</span>
  1013→    }
  1014→    return String(value)
  1015→  }
  1016→
  1017→  return (
  1018→    <div className="rounded-md border border-zinc-800 bg-zinc-950/50">
