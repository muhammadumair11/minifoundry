    69→// ─── Object type card ─────────────────────────────────────────────────
    70→
    71→function TypeCard({
    72→  type,
    73→  onRequestDelete,
    74→}: {
    75→  type: OntologyObjectType
    76→  onRequestDelete: (t: OntologyObjectType) => void
    77→}) {
    78→  const router = useRouter()
    79→  const color = getObjectColorClasses(type.color)
    80→
    81→  const titleProp = type.properties.find((p) => p.isTitle)
    82→  const previewProps = [titleProp, ...type.properties.filter((p) => !p.isTitle)]
    83→    .filter((p): p is NonNullable<typeof p> => !!p)
    84→    .slice(0, 3)
    85→
    86→  const open = () => router.push(`/objects?type=${encodeURIComponent(type.apiName)}`)
    87→
    88→  return (
    89→    <div
    90→      role="link"
    91→      tabIndex={0}
    92→      aria-label={`Open ${type.displayName} objects`}
    93→      onClick={open}
    94→      onKeyDown={(e) => {
    95→        if (e.key === 'Enter' || e.key === ' ') {
    96→          e.preventDefault()
    97→          open()
    98→        }
    99→      }}
   100→      className="group relative flex cursor-pointer flex-col rounded-lg border border-border bg-card p-4 outline-none transition-colors hover:bg-accent focus-visible:border-foreground/40"
   101→    >
   102→      {/* Icon + names + menu */}
   103→      <div className="flex items-start gap-3">
   104→        <span
   105→          className={cn(
   106→            'flex h-10 w-10 shrink-0 items-center justify-center rounded-lg border',
   107→            color.border,
   108→            color.bg,
   109→          )}
   110→        >
   111→          <OntologyIcon name={type.icon} className={cn('h-5 w-5', color.text)} />
   112→        </span>
   113→        <div className="min-w-0 flex-1">
   114→          <p className="truncate text-sm font-semibold text-foreground">{type.displayName}</p>
   115→          <p className="truncate font-mono text-[11px] text-muted-foreground">{type.apiName}</p>
   116→        </div>
   117→        <span
   118→          className="shrink-0"
   119→          onClick={(e) => e.stopPropagation()}
   120→          onKeyDown={(e) => e.stopPropagation()}
   121→          role="presentation"
   122→        >
   123→          <DropdownMenu>
   124→            <DropdownMenuTrigger asChild>
   125→              <Button
   126→                variant="ghost"
   127→                size="sm"
   128→                className="h-7 w-7 p-0 text-muted-foreground hover:bg-accent hover:text-foreground"
   129→                aria-label={`Options for ${type.displayName}`}
   130→              >
   131→                <MoreHorizontal className="h-4 w-4" aria-hidden />
   132→              </Button>
   133→            </DropdownMenuTrigger>
   134→            <DropdownMenuContent align="end">
   135→              <DropdownMenuItem
   136→                onSelect={() => open()}
   137→                className="text-xs text-foreground"
   138→              >
   139→                <ExternalLink className="mr-2 h-3.5 w-3.5" aria-hidden />
   140→                Open objects
   141→              </DropdownMenuItem>
   142→              <DropdownMenuSeparator />
   143→              <DropdownMenuItem
   144→                onSelect={() => onRequestDelete(type)}
   145→                className="text-xs text-rose-600 dark:text-rose-400 hover:!bg-rose-500/10 hover:!text-rose-600"
   146→              >
   147→                <Trash2 className="mr-2 h-3.5 w-3.5" aria-hidden />
   148→                Delete type…
   149→              </DropdownMenuItem>
   150→            </DropdownMenuContent>
   151→          </DropdownMenu>
   152→        </span>
   153→      </div>
   154→
   155→      {/* Counts + backing dataset */}
   156→      <div className="mt-3 flex items-baseline justify-between gap-2">
   157→        <p className="font-mono text-xs text-foreground">
   158→          {type.objectCount ?? '—'} <span className="text-muted-foreground">objects</span>
   159→        </p>
   160→        {type.sourceDatasetId ? (
   161→          <Link
   162→            href={`/datasets/${type.sourceDatasetId}`}
   163→            onClick={(e) => e.stopPropagation()}
   164→            className="truncate font-mono text-[11px] text-foreground hover:underline underline-offset-2"
   165→            title={`Backing dataset — ${type.sourceDatasetName ?? 'dataset'}`}
   166→          >
   167→            {type.sourceDatasetName ?? 'dataset'}
   168→          </Link>
   169→        ) : (
   170→          <span className="font-mono text-[11px] text-muted-foreground/80">no dataset</span>
   171→        )}
   172→      </div>
   173→
   174→      {/* Property / action chips */}
   175→      <div className="mt-2 flex flex-wrap items-center gap-1.5">
   176→        <span className="rounded border border-border bg-muted px-1.5 py-0.5 font-mono text-[10px] text-muted-foreground">
   177→          {type.properties.length} props
   178→        </span>
   179→        <span className="rounded border border-border bg-muted px-1.5 py-0.5 font-mono text-[10px] text-muted-foreground">
   180→          {type.actions.length} actions
   181→        </span>
   182→        {type.linksOut.length + type.linksIn.length > 0 && (
   183→          <span className="rounded border border-border bg-muted px-1.5 py-0.5 font-mono text-[10px] text-muted-foreground">
   184→            {type.linksOut.length + type.linksIn.length} links
   185→          </span>
   186→        )}
   187→      </div>
   188→
   189→      {/* Preview properties */}
   190→      <ul className="mt-3 space-y-1 border-t border-border pt-2.5">
   191→        {previewProps.map((p) => (
   192→          <li key={p.apiName} className="truncate font-mono text-[11px]">
   193→            <span className={cn(p.isTitle ? 'font-semibold text-foreground' : 'text-muted-foreground')}>
   194→              {p.displayName}
   195→            </span>
   196→            <span className="ml-1.5 text-muted-foreground/70">{p.dataType}</span>
   197→          </li>
   198→        ))}
   199→      </ul>
   200→
   201→      {/* Links summary */}
   202→      <p className="mt-2.5 font-mono text-[10px] text-muted-foreground/80">
   203→        {type.linksOut.length + type.linksIn.length > 0 ? (
   204→          <>
   205→            <span className="text-muted-foreground">→ {type.linksOut.length} outgoing</span>
   206→            {' · '}
   207→            <span className="text-muted-foreground">{type.linksIn.length} incoming</span>
   208→          </>
   209→        ) : (
   210→          'no links'
   211→        )}
   212→      </p>
   213→    </div>
   214→  )
   215→}

   217→// ─── Page ─────────────────────────────────────────────────────────────
   218→
   219→export default function OntologyPage() {
   220→  const queryClient = useQueryClient()
   221→  const [deleting, setDeleting] = useState<OntologyObjectType | null>(null)
   222→
   223→  const ontologyQuery = useQuery<OntologyListResponse>({
   224→    queryKey: ['ontology'],
   225→    queryFn: () => apiJson<OntologyListResponse>('/api/ontology'),
   226→  })
   227→  const types = useMemo(() => ontologyQuery.data?.objectTypes ?? [], [ontologyQuery.data])
   228→
   229→  const actionsQuery = useQuery<ActionsListResponse>({
   230→    queryKey: ['actions'],
   231→    queryFn: () => apiJson<ActionsListResponse>('/api/actions'),
   232→  })
   233→  const actions = useMemo(() => actionsQuery.data?.actions ?? [], [actionsQuery.data])
   234→
   235→  const logsQuery = useQuery<ActionLogsResponse>({
   236→    queryKey: ['action-logs'],
   237→    queryFn: () => apiJson<ActionLogsResponse>('/api/actions/logs?limit=10'),
   238→    refetchInterval: 15_000,
   239→  })
   240→  const logs = useMemo(() => logsQuery.data?.logs ?? [], [logsQuery.data])
   241→
   242→  // All links across types (deduped — linksIn mirror other types' linksOut).
   243→  const allLinks = useMemo(() => {
   244→    const rows: { id: string; from: OntologyObjectType; link: OutgoingLinkSummary }[] = []
   245→    const seen = new Set<string>()
   246→    for (const t of types) {
   247→      for (const l of t.linksOut) {
   248→        if (seen.has(l.id)) continue
   249→        seen.add(l.id)
   250→        rows.push({ id: l.id, from: t, link: l })
   251→      }
   252→    }
   253→    return rows
   254→  }, [types])
   255→
   256→  const seedMutation = useMutation({
   257→    mutationFn: () => apiJson<SeedOntologyResponse>('/api/ontology/seed-demo', { method: 'POST' }),
   258→    onSuccess: (res) => {
   259→      if (res.seeded) {
   260→        toast.success('Demo ontology seeded', {
   261→          description: `Created ${res.created.length} object types: ${res.created.join(', ')}`,
   262→        })
   263→      } else {
   264→        toast.info('Demo ontology already present', { description: 'Nothing new to seed.' })
   265→      }
   266→      void queryClient.invalidateQueries({ queryKey: ['ontology'] })
   267→      void queryClient.invalidateQueries({ queryKey: ['stats'] })
   268→    },
   269→    onError: (err: Error) => toast.error('Seeding failed', { description: err.message }),
   270→  })
   271→
   272→  const deleteMutation = useMutation({
   273→    mutationFn: (t: OntologyObjectType) =>
   274→      apiJson<void>(`/api/ontology/${t.id}`, { method: 'DELETE' }),
   275→    onSuccess: (_res, t) => {
   276→      toast.success('Object type deleted', {
   277→        description: `${t.displayName} removed from the ontology (links + actions cascaded).`,
   278→      })
   279→      void queryClient.invalidateQueries({ queryKey: ['ontology'] })
   280→      void queryClient.invalidateQueries({ queryKey: ['actions'] })
   281→      void queryClient.invalidateQueries({ queryKey: ['action-logs'] })
   282→      void queryClient.invalidateQueries({ queryKey: ['stats'] })
   283→      setDeleting(null)
   284→    },
   285→    onError: (err: Error) =>
   286→      toast.error('Could not delete object type', { description: err.message }),
   287→  })
   288→
   289→  const loading = ontologyQuery.isLoading
   290→
   291→  return (
   292→    <div className="mx-auto w-full max-w-6xl px-4 py-6 sm:py-8">
   293→      {/* a) Header (compact Foundry pattern) */}
   294→      <header className="mb-4 flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   295→        <div className="flex min-w-0 flex-wrap items-center gap-2">
   296→          <Network className="h-4 w-4 shrink-0 text-muted-foreground" aria-hidden />
   297→          <h1 className="text-sm font-medium tracking-tight text-foreground">Ontology</h1>
   298→          <span className="text-xs text-muted-foreground">
   299→            the semantic layer — objects, links, write-back actions
   300→          </span>
   301→          {types.length > 0 && (
   302→            <span className="font-mono text-[11px] text-muted-foreground/80">
   303→              {types.length} types · {allLinks.length} links · {actions.length} actions
   304→            </span>
   305→          )}
   306→        </div>
   307→        <div className="flex shrink-0 flex-wrap items-center gap-2">
   308→          <Button
   309→            size="sm"
   310→            variant="outline"
   311→            disabled={types.length >= 3 || seedMutation.isPending}
   312→            onClick={() => seedMutation.mutate()}
   313→            className="h-8 gap-1.5 text-xs"
   314→            title={
   315→              types.length >= 3 ? 'Demo ontology already seeded' : 'Create the supply-chain demo ontology'
   316→            }
   317→          >
   318→            <Sparkles
   319→              className={cn('h-3.5 w-3.5', seedMutation.isPending && 'animate-pulse')}
   320→              aria-hidden
   321→            />
   322→            Seed demo ontology
   323→          </Button>
   324→          <NewObjectTypeDialog />
   325→        </div>
   326→      </header>
   327→
   328→      {/* b) Object type cards */}
   329→      {loading ? (
   330→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3">
   331→          {[0, 1, 2].map((i) => (
   332→            <Skeleton key={i} className="h-56 w-full" />
   333→          ))}
   334→        </div>
   335→      ) : ontologyQuery.isError ? (
   336→        <Card className="border-border bg-card">
   337→          <CardContent className="p-6 text-center text-sm text-rose-600 dark:text-rose-400">
   338→            {ontologyQuery.error instanceof Error
   339→              ? ontologyQuery.error.message
   340→              : 'Failed to load the ontology.'}
   341→          </CardContent>
   342→        </Card>
   343→      ) : types.length === 0 ? (
   344→        <Card className="border-dashed">
   345→          <CardContent className="flex flex-col items-center gap-3 p-8 text-center">
   346→            <Network className="h-8 w-8 text-muted-foreground" aria-hidden />
   347→            <div>
   348→              <p className="text-base font-medium text-foreground">No object types yet</p>
   349→              <p className="mx-auto mt-1 max-w-md text-sm text-muted-foreground">
   350→                Seed the supply-chain demo (shipment, supplier, warehouse with links + actions) or
   351→                map your own dataset into an object type.
   352→              </p>
   353→            </div>
   354→          </CardContent>
   355→        </Card>
   356→      ) : (
   357→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3">
   358→          {types.map((t) => (
   359→            <TypeCard key={t.id} type={t} onRequestDelete={setDeleting} />
   360→          ))}
   361→        </div>
   362→      )}
   363→
   364→      {/* c) Links across types */}
   365→      <Card className="mt-4 border-border bg-card">
   366→        <CardHeader className="flex-row items-center justify-between gap-2 border-b border-border p-4">
   367→          <div className="flex items-center gap-2">
   368→            <Link2 className="h-4 w-4 text-muted-foreground" aria-hidden />
   369→            <CardTitle className="text-sm font-semibold text-foreground">Links</CardTitle>
   370→            {!loading && (
   371→              <span className="rounded border border-border bg-muted px-1.5 py-0.5 font-mono text-[10px] text-muted-foreground">
   372→                {allLinks.length}
   373→              </span>
   374→            )}
   375→          </div>
   376→        </CardHeader>
   377→        <CardContent className="p-0">
   378→          {loading ? (
   379→            <div className="space-y-2 p-4">
   380→              {[0, 1].map((i) => (
   381→                <Skeleton key={i} className="h-10 w-full" />
   382→              ))}
   383→            </div>
   384→          ) : allLinks.length === 0 ? (
   385→            <div className="p-6 text-center text-sm text-muted-foreground">
   386→              No links yet — links connect object types (e.g. supplier → shipment via a foreign
   387→              key) and are created in the ontology backend.
   388→            </div>
   389→          ) : (
   390→            <div className="foundry-scroll max-h-96 overflow-auto">
   391→              <Table>
   392→                <TableHeader className="sticky top-0 z-10 bg-card">
   393→                  <TableRow className="border-border hover:bg-transparent">
   394→                    <TableHead className="h-9 px-4 text-xs text-muted-foreground">From → To</TableHead>
   395→                    <TableHead className="h-9 px-3 text-xs text-muted-foreground">Cardinality</TableHead>
   396→                    <TableHead className="h-9 px-3 text-xs text-muted-foreground">Link</TableHead>
   397→                  </TableRow>
   398→                </TableHeader>
   399→                <TableBody>
   400→                  {allLinks.map(({ id, from, link }) => (
   401→                    <TableRow key={id} className="border-border">
   402→                      <TableCell className="px-4 py-2">
   403→                        <span className="flex flex-wrap items-center gap-1.5 text-sm">
   404→                          <span className="font-medium text-foreground">{from.displayName}</span>
   405→                          <ArrowRight className="h-3.5 w-3.5 text-muted-foreground" aria-hidden />
   406→                          <span className="font-medium text-foreground">{link.toTypeDisplayName}</span>
   407→                        </span>
   408→                      </TableCell>
   409→                      <TableCell className="px-3 py-2">
   410→                        <Badge
   411→                          variant="outline"
   412→                          className={cn('font-mono text-[10px]', getCardinalityClasses(link.cardinality))}
   413→                        >
   414→                          {getCardinalityLabel(link.cardinality)}
   415→                        </Badge>
   416→                      </TableCell>
   417→                      <TableCell className="px-3 py-2 font-mono text-xs text-muted-foreground">
   418→                        {link.apiName}
   419→                      </TableCell>
   420→                    </TableRow>
   421→                  ))}
   422→                </TableBody>
   423→              </Table>
   424→            </div>
   425→          )}
   426→        </CardContent>
   427→      </Card>
   428→
   429→      {/* d) Actions */}
   430→      <Card className="mt-4 border-border bg-card">
   431→        <CardHeader className="flex-row items-center justify-between gap-2 border-b border-border p-4">
   432→          <div className="flex items-center gap-2">
   433→            <Zap className="h-4 w-4 text-muted-foreground" aria-hidden />
   434→            <CardTitle className="text-sm font-semibold text-foreground">Actions</CardTitle>
   435→            {!actionsQuery.isLoading && (
   436→              <span className="rounded border border-border bg-muted px-1.5 py-0.5 font-mono text-[10px] text-muted-foreground">
   437→                {actions.length}
   438→              </span>
   439→            )}
   440→          </div>
   441→        </CardHeader>
   442→        <CardContent className="p-0">
   443→          {actionsQuery.isLoading ? (
   444→            <div className="space-y-2 p-4">
   445→              {[0, 1].map((i) => (
   446→                <Skeleton key={i} className="h-10 w-full" />
   447→              ))}
   448→            </div>
   449→          ) : actionsQuery.isError ? (
   450→            <p className="p-6 text-center text-sm text-rose-600 dark:text-rose-400">
   451→              {actionsQuery.error instanceof Error
   452→                ? actionsQuery.error.message
   453→                : 'Failed to load actions.'}
   454→            </p>
   455→          ) : actions.length === 0 ? (
   456→            <div className="p-6 text-center text-sm text-muted-foreground">
   457→              No actions yet — actions are the write-back layer (create / update / delete objects
   458→              as new dataset versions).
   459→            </div>
   460→          ) : (
   461→            <div className="foundry-scroll max-h-96 overflow-auto">
   462→              <Table>
   463→                <TableHeader className="sticky top-0 z-10 bg-card">
   464→                  <TableRow className="border-border hover:bg-transparent">
   465→                    <TableHead className="h-9 px-4 text-xs text-muted-foreground">Action</TableHead>
   466→                    <TableHead className="h-9 px-3 text-xs text-muted-foreground">Operation</TableHead>
   467→                    <TableHead className="h-9 px-3 text-xs text-muted-foreground">Target</TableHead>
   468→                    <TableHead className="h-9 px-3 text-xs text-muted-foreground">Params</TableHead>
   469→                    <TableHead className="h-9 px-3 text-xs text-muted-foreground">Enabled</TableHead>
   470→                    <TableHead className="h-9 px-3 text-right text-xs text-muted-foreground">
   471→                      <span className="sr-only">Open</span>
   472→                    </TableHead>
   473→                  </TableRow>
   474→                </TableHeader>
   475→                <TableBody>
   476→                  {actions.map((a: ActionSummary) => {
   477→                    const op = getOperationClasses(a.operation)
   478→                    return (
   479→                      <TableRow key={a.id} className="border-border">
   480→                        <TableCell className="max-w-[260px] px-4 py-2">
   481→                          <p className="truncate text-sm font-medium text-foreground">
   482→                            {a.displayName}
   483→                          </p>
   484→                          <p className="truncate font-mono text-[11px] text-muted-foreground">{a.apiName}</p>
   485→                        </TableCell>
   486→                        <TableCell className="px-3 py-2">
   487→                          <Badge variant="outline" className={cn('text-[10px]', op.badge)}>
   488→                            {a.operation}
   489→                          </Badge>
   490→                        </TableCell>
   491→                        <TableCell className="px-3 py-2">
   492→                          <Link
   493→                            href={`/objects?type=${encodeURIComponent(a.objectTypeApiName)}`}
   494→                            className="text-sm text-foreground hover:underline underline-offset-2"
   495→                          >
   496→                            {a.objectTypeDisplayName}
   497→                          </Link>
   498→                        </TableCell>
   499→                        <TableCell className="px-3 py-2 font-mono text-xs text-foreground">
   500→                          {a.paramCount}
   501→                        </TableCell>
   502→                        <TableCell className="px-3 py-2">
   503→                          <Badge
   504→                            variant="outline"
   505→                            className={cn(
   506→                              'text-[10px]',
   507→                              a.enabled
   508→                                ? 'border-emerald-500/40 bg-emerald-500/10 text-emerald-600 dark:text-emerald-400'
   509→                                : 'border-border bg-muted text-muted-foreground',
   510→                            )}
   511→                          >
   512→                            {a.enabled ? 'enabled' : 'disabled'}
   513→                          </Badge>
   514→                        </TableCell>
