    55→function useDebouncedValue<T>(value: T, delayMs: number): T {
    56→  const [debounced, setDebounced] = useState(value)
    57→  useEffect(() => {
    58→    const t = setTimeout(() => setDebounced(value), delayMs)
    59→    return () => clearTimeout(t)
    60→  }, [value, delayMs])
    61→  return debounced
    62→}

    64→/** Compact value renderer for object cells / property rows. */
    65→function renderValue(value: unknown): React.ReactNode {
    66→  if (value === null || value === undefined || value === '') {
    67→    return <span className="text-zinc-700">∅</span>
    68→  }
    69→  if (value === MASK_LITERAL) {
    70→    return <span className="italic text-zinc-600">{MASK_LITERAL}</span>
    71→  }
    72→  if (typeof value === 'boolean') {
    73→    return <span className={value ? 'text-amber-400' : 'text-zinc-500'}>{String(value)}</span>
    74→  }
    75→  if (typeof value === 'number') {
    76→    return <span className="tabular-nums">{value.toLocaleString()}</span>
    77→  }
    78→  return String(value)
    79→}

    83→function ActiveActionDialog({
    84→  actionApiName,
    85→  prefillPk,
    86→  onClose,
    87→}: {
    88→  actionApiName: string
    89→  prefillPk: string | null
    90→  onClose: () => void
    91→}) {
    92→  const actionQuery = useQuery<ActionResponse>({
    93→    queryKey: ['action', actionApiName],
    94→    queryFn: () => apiJson<ActionResponse>(`/api/actions/${encodeURIComponent(actionApiName)}`),
    95→    enabled: actionApiName !== '',
    96→  })
    97→
    98→  useEffect(() => {
    99→    if (actionQuery.isError) {
   100→      toast.error('Could not load action', {
   101→        description:
   102→          actionQuery.error instanceof Error ? actionQuery.error.message : 'Unknown error.',
   103→      })
   104→      onClose()
   105→    }
   106→  }, [actionQuery.isError, actionQuery.error, onClose])
   107→
   108→  if (!actionQuery.data) return null
   109→
   110→  return (
   111→    <ActionDialog
   112→      action={actionQuery.data.action}
   113→      open
   114→      onOpenChange={(next) => {
   115→        if (!next) onClose()
   116→      }}
   117→      prefillPk={prefillPk}
   118→    />
   119→  )
   120→}

   329→function WorkshopView() {
   330→  const routeParams = useParams<{ type: string }>()
   331→  const typeApiName = routeParams?.type ?? ''
   332→  const searchParams = useSearchParams()
   333→  const actionParam = searchParams.get('action')
   334→  const pkParam = searchParams.get('pk')
   335→
   336→  const [searchInput, setSearchInput] = useState('')
   337→  const [page, setPage] = useState(1)
   338→  const [selectedPk, setSelectedPk] = useState<string | null>(null)
   339→  const [activeActionApiName, setActiveActionApiName] = useState<string | null>(null)
   340→
   341→  const debouncedSearch = useDebouncedValue(searchInput, 400)
   342→
   343→  // ?pk= → preselect (and type change → reset) by syncing during render.
   344→  const [pkSync, setPkSync] = useState('')
   345→  const pkSignal = `${typeApiName}\u0000${pkParam ?? ''}`
   346→  if (pkSync !== pkSignal) {
   347→    setPkSync(pkSignal)
   348→    setSelectedPk(pkParam)
   349→  }
   350→
   351→  // ?action= → auto-open the action dialog once per param value (closing it
   352→  // does not re-open while the URL keeps the same ?action=).
   353→  const [consumedActionParam, setConsumedActionParam] = useState<string | null>(null)
   354→  if (actionParam && actionParam !== consumedActionParam) {
   355→    setConsumedActionParam(actionParam)
   356→    setActiveActionApiName(actionParam)
   357→  }
   358→
   359→  // Ontology list — type meta + enabled actions.
   360→  const ontologyQuery = useQuery<OntologyListResponse>({
   361→    queryKey: ['ontology'],
   362→    queryFn: () => apiJson<OntologyListResponse>('/api/ontology'),
   363→  })
   364→  const typeInfo = useMemo(
   365→    () => ontologyQuery.data?.objectTypes.find((t) => t.apiName === typeApiName) ?? null,
   366→    [ontologyQuery.data, typeApiName],
   367→  )
   368→  const enabledActions = useMemo(
   369→    () => (typeInfo?.actions ?? []).filter((a) => a.enabled),
   370→    [typeInfo],
   371→  )
   372→
   373→  // Object table.
   374→  const objectsQuery = useQuery<ObjectsListResponse>({
   375→    queryKey: ['objects', typeApiName, debouncedSearch, page],
   376→    queryFn: () =>
   377→      apiJson<ObjectsListResponse>(
   378→        `/api/objects?type=${encodeURIComponent(typeApiName)}&query=${encodeURIComponent(debouncedSearch)}&page=${page}&pageSize=${PAGE_SIZE}`,
   379→      ),
   380→    enabled: typeApiName !== '',
   381→    placeholderData: (prev) => prev,
   382→  })
   383→  const rows = objectsQuery.data?.rows ?? []
   384→  const allColumns = objectsQuery.data?.columns ?? []
   385→  const pkColumn = objectsQuery.data?.objectType.primaryKey ?? null
   386→  // The table already renders a dedicated pk column — drop the property that mirrors it.
   387→  const tableColumns = useMemo(
   388→    () => allColumns.filter((c) => !c.hideInTable && c.apiName !== pkColumn),
   389→    [allColumns, pkColumn],
   390→  )
   391→  const total = objectsQuery.data?.total ?? 0
   392→  const totalPages = Math.max(1, Math.ceil(total / (objectsQuery.data?.pageSize ?? PAGE_SIZE)))
   393→  const titleProperty = objectsQuery.data?.objectType.titleProperty ?? null
   394→
   395→  const unknownType = !ontologyQuery.isLoading && !!ontologyQuery.data && !typeInfo
   396→
   397→  return (
   398→    <div className="mx-auto w-full max-w-6xl px-4 py-6 sm:py-8">
   399→      {/* a) Type header */}
   400→      <header className="mb-5">
   401→        {ontologyQuery.isLoading ? (
   402→          <div className="flex items-center gap-3">
   403→            <Skeleton className="h-10 w-10 bg-zinc-800" />
   404→            <div className="space-y-1.5">
   405→              <Skeleton className="h-6 w-40 bg-zinc-800" />
   406→              <Skeleton className="h-3.5 w-56 bg-zinc-800" />
   407→            </div>
   408→          </div>
   409→        ) : typeInfo ? (
   410→          <div className="flex flex-col gap-3 sm:flex-row sm:items-start sm:justify-between">
   411→            <div className="flex items-start gap-3">
   412→              <TypeIconBadge type={typeInfo} />
   413→              <div>
   414→                <div className="flex flex-wrap items-baseline gap-2">
   415→                  <h1 className="text-xl font-semibold tracking-tight text-zinc-100 sm:text-2xl">
   416→                    {typeInfo.displayName}
   417→                  </h1>
   418→                  <span className="font-mono text-xs text-zinc-500">{typeInfo.apiName}</span>
   419→                </div>
   420→                <p className="mt-1 flex flex-wrap items-center gap-x-3 gap-y-1 font-mono text-[11px] text-zinc-500">
   421→                  <span>
   422→                    {typeInfo.objectCount ?? '—'} <span className="text-zinc-600">objects</span>

   539→              {objectsQuery.isLoading ? (
   540→                <div className="space-y-2 p-4">
   541→                  {[0, 1, 2, 3, 4].map((i) => (
   542→                    <Skeleton key={i} className="h-9 w-full bg-zinc-800" />
   543→                  ))}
   544→                </div>
   545→              ) : objectsQuery.isError ? (
   546→                <p className="p-6 text-center text-sm text-rose-400">
   547→                  {objectsQuery.error instanceof Error
   548→                    ? objectsQuery.error.message
   549→                    : 'Failed to load objects.'}
   550→                </p>
   551→              ) : rows.length === 0 ? (
   552→                <div className="p-6 text-center text-sm text-zinc-500">
   553→                  {debouncedSearch
   554→                    ? `No objects match “${debouncedSearch}”.`
   555→                    : 'No objects in the backing dataset yet.'}
   556→                </div>
   557→              ) : (
   558→                <div className="foundry-scroll max-h-[520px] overflow-auto">
   559→                  <Table>
   560→                    <TableHeader className="sticky top-0 z-10 bg-zinc-900">
   561→                      <TableRow className="border-zinc-800 hover:bg-transparent">
   562→                        <TableHead className="h-8 px-4 font-mono text-[10px] uppercase tracking-wider text-zinc-500">
   563→                          {objectsQuery.data?.objectType.primaryKey ?? 'pk'}
   564→                        </TableHead>
   565→                        {tableColumns.map((c) => (
   566→                          <TableHead
   567→                            key={c.apiName}
   568→                            className="h-8 whitespace-nowrap px-3 font-mono text-[10px] uppercase tracking-wider text-zinc-500"
   569→                          >
   570→                            {c.displayName}
   571→                          </TableHead>
   572→                        ))}
   573→                      </TableRow>
   574→                    </TableHeader>
   575→                    <TableBody>
   576→                      {rows.map((row) => {
   577→                        const selected = row.pk === selectedPk
   578→                        return (
   579→                          <TableRow
   580→                            key={row.pk}
   581→                            aria-selected={selected}
   582→                            onClick={() => setSelectedPk(row.pk)}
   583→                            className={cn(
   584→                              'cursor-pointer border-zinc-800/70',
   585→                              selected
   586→                                ? 'border-l-2 border-l-emerald-400 bg-zinc-800/40 hover:bg-zinc-800/40'
   587→                                : 'hover:bg-zinc-800/30',
   588→                            )}
   589→                          >
   590→                            <TableCell className="px-4 py-2 font-mono text-xs text-zinc-500">
   591→                              {row.pk}
   592→                            </TableCell>
   593→                            {tableColumns.map((c) => (
   594→                              <TableCell key={c.apiName} className="max-w-[220px] px-3 py-2">
   595→                                {c.isTitle ? (
   596→                                  <button
   597→                                    type="button"
   598→                                    onClick={(e) => {
   599→                                      e.stopPropagation()
   600→                                      setSelectedPk(row.pk)
   601→                                    }}
   602→                                    className="max-w-full truncate text-left text-xs font-semibold text-emerald-400 hover:underline"
   603→                                    title={`Select ${row.pk}`}
   604→                                  >
   605→                                    {renderValue(row.values[c.apiName])}
   606→                                  </button>
   607→                                ) : (
   608→                                  <span className="block max-w-[220px] truncate font-mono text-xs text-zinc-300">
   609→                                    {renderValue(row.values[c.apiName])}
   610→                                  </span>
   611→                                )}
   612→                              </TableCell>
   613→                            ))}
   614→                          </TableRow>
   615→                        )
   616→                      })}
   617→                    </TableBody>
   618→                  </Table>
   619→                </div>
   620→              )}
   621→
   622→              {/* Pagination footer */}
   623→              {!objectsQuery.isLoading && rows.length > 0 && (
   624→                <div className="flex items-center justify-between gap-2 border-t border-zinc-800 px-4 py-2.5">
   625→                  <p className="font-mono text-[11px] text-zinc-500">
   626→                    {total} objects · page {page} / {totalPages}
   627→                  </p>
   628→                  <div className="flex items-center gap-1.5">
   629→                    <Button
   630→                      size="sm"
   631→                      variant="outline"
   632→                      className="h-7 border-zinc-800 bg-zinc-950/60 px-2 text-zinc-300 hover:bg-zinc-800 hover:text-zinc-100"
   633→                      disabled={page <= 1}
   634→                      onClick={() => setPage((p) => Math.max(1, p - 1))}
   635→                      aria-label="Previous page"
   636→                    >
   637→                      <ChevronLeft className="h-3.5 w-3.5" aria-hidden />
   638→                    </Button>
   639→                    <Button
   640→                      size="sm"
   641→                      variant="outline"
   642→                      className="h-7 border-zinc-800 bg-zinc-950/60 px-2 text-zinc-300 hover:bg-zinc-800 hover:text-zinc-100"
   643→                      disabled={page >= totalPages}
   644→                      onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   645→                      aria-label="Next page"
   646→                    >
   647→                      <ChevronRight className="h-3.5 w-3.5" aria-hidden />
   648→                    </Button>
   649→                  </div>
   650→                </div>
   651→              )}
   652→            </CardContent>
   653→          </Card>
   654→
   655→          {/* c) RIGHT: detail panel */}
   656→          <div className="w-full shrink-0 lg:sticky lg:top-[72px] lg:w-96">
   657→            <Card className="border-zinc-800 bg-zinc-900/60">
   658→              {selectedPk ? (
   659→                <ObjectDetailPanel
   660→                  typeApiName={typeApiName}
   661→                  selectedPk={selectedPk}
   662→                  columns={allColumns}
   663→                  titleProperty={titleProperty}
   664→                  onClear={() => setSelectedPk(null)}
   665→                />
   666→              ) : (
   667→                <CardContent className="flex flex-col items-center gap-2 p-8 text-center">
   668→                  <Compass className="h-7 w-7 text-zinc-600" aria-hidden />
   669→                  <p className="text-sm font-medium text-zinc-400">No object selected</p>
   670→                  <p className="text-xs text-zinc-600">
   671→                    Click a row to inspect its properties and links.
   672→                  </p>
   673→                </CardContent>
   674→              )}
   675→            </Card>
   676→          </div>
   677→        </div>
   678→      )}
   679→
   680→      {/* d) Action dialog (dropdown picks + ?action= deep link) */}
   681→      {activeActionApiName && (
   682→        <ActiveActionDialog
   683→          actionApiName={activeActionApiName}
   684→          prefillPk={selectedPk}
   685→          onClose={() => setActiveActionApiName(null)}
   686→        />
   687→      )}
   688→    </div>
   689→  )
   690→}
