    76→function useDebouncedValue<T>(value: T, delayMs: number): T {
    77→  const [debounced, setDebounced] = useState(value)
    78→  useEffect(() => {
    79→    const t = setTimeout(() => setDebounced(value), delayMs)
    80→    return () => clearTimeout(t)
    81→  }, [value, delayMs])
    82→  return debounced
    83→}

    85→function renderCellValue(value: unknown, type: string | undefined): React.ReactNode {
    86→  if (value === null || value === undefined || value === '') {
    87→    return <span className="text-muted-foreground/60">∅</span>
    88→  }
    89→  if (value === MASK_LITERAL) {
    90→    return <span className="italic text-muted-foreground/80">{MASK_LITERAL}</span>
    91→  }
    92→  if (typeof value === 'boolean' || type === 'boolean') {
    93→    return (
    94→      <span className={value ? 'text-amber-600 dark:text-amber-400' : 'text-muted-foreground'}>
    95→        {String(value)}
    96→      </span>
    97→    )
    98→  }
    99→  if (typeof value === 'number') {
   100→    return <span className="tabular-nums">{value.toLocaleString()}</span>
   101→  }
   102→  return String(value)
   103→}

   521→            ) : (rowsQuery.data?.rows.length ?? 0) === 0 ? (
   522→              <div className="p-8 text-center text-sm text-muted-foreground">
   523→                {searchQuery.trim()
   524→                  ? `No rows match “${searchQuery}”.`
   525→                  : 'This version contains no rows.'}
   526→              </div>
   527→            ) : (
   528→              <div className="foundry-scroll max-h-[520px] overflow-auto">
   529→                <Table>
   530→                  <TableHeader className="sticky top-0 z-10 bg-card">
   531→                    <TableRow className="border-border hover:bg-transparent">
   532→                      {columns.map((col) => (
   533→                        <TableHead
   534→                          key={col.name}
   535→                          className={cn(
   536→                            'h-8 whitespace-nowrap px-3 font-mono text-[11px] text-muted-foreground',
   537→                            col.type === 'number' && 'text-right',
   538→                          )}
   539→                        >
   540→                          {col.name}
   541→                          <span className="ml-1 text-muted-foreground/70">
   542→                            {col.type === 'number' ? '#' : col.type === 'boolean' ? '⌥' : 'A'}
   543→                          </span>
   544→                        </TableHead>
   545→                      ))}
   546→                    </TableRow>
   547→                  </TableHeader>
   548→                  <TableBody>
   549→                    {rowsQuery.data?.rows.map((row, i) => (
   550→                      <TableRow key={i} className="border-border hover:bg-accent/50">
   551→                        {columns.map((col) => (
   552→                          <TableCell
   553→                            key={col.name}
   554→                            className={cn(
   555→                              'whitespace-nowrap px-3 py-1.5 font-mono text-xs text-foreground',
   556→                              col.type === 'number' && 'text-right tabular-nums',
   557→                            )}
   558→                          >
   559→                            {renderCellValue(row[col.name], col.type)}
   560→                          </TableCell>
   561→                        ))}
   562→                      </TableRow>
   563→                    ))}
   564→                  </TableBody>
   565→                </Table>
   566→              </div>
   567→            )}
   568→
   569→            {/* Pagination footer */}
   570→            <div className="flex items-center justify-between gap-2 border-t border-border p-3">
   571→              <Button
   572→                variant="outline"
   573→                size="sm"
   574→                disabled={page <= 1 || rowsQuery.isFetching}
   575→                onClick={() => setPage((p) => Math.max(1, p - 1))}
   576→                className="h-8 text-xs"
   577→                aria-label="Previous page"
   578→              >
   579→                <ChevronLeft className="h-3.5 w-3.5" aria-hidden />
   580→                Prev
   581→              </Button>
   582→              <p className="font-mono text-[11px] text-muted-foreground">
   583→                page {rowsQuery.data?.page ?? page} of {totalPages} · {total.toLocaleString()} rows
   584→              </p>
   585→              <Button
   586→                variant="outline"
   587→                size="sm"
   588→                disabled={page >= totalPages || rowsQuery.isFetching}
   589→                onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   590→                className="h-8 text-xs"
   591→                aria-label="Next page"
   592→              >
   593→                Next
   594→                <ChevronRight className="h-3.5 w-3.5" aria-hidden />
   595→              </Button>
   596→            </div>
   597→          </Card>
   598→        </TabsContent>
   599→
   600→        {/* ── Schema ── */}
   601→        <TabsContent value="schema" className="mt-3">
   602→          <Card>
   603→            <CardContent className="p-0">
   604→              {columns.length === 0 ? (
   605→                <p className="p-6 text-center text-sm text-muted-foreground">No schema recorded.</p>
   606→              ) : (
   607→                <ul className="divide-y divide-border">
   608→                  {columns.map((col) => (
   609→                    <li key={col.name} className="flex items-center justify-between gap-3 px-4 py-2">
   610→                      <span className="truncate font-mono text-xs text-foreground">{col.name}</span>
   611→                      <span className="flex shrink-0 items-center gap-1.5">
   612→                        {col.markedWith && (
   613→                          <MarkingChip
   614→                            name={col.markedWith}
   615→                            color={markingColorByName.get(col.markedWith)}
   616→                          />
   617→                        )}
   618→                        <Badge
   619→                          variant="outline"
   620→                          className="shrink-0 border-border bg-muted font-mono text-[10px] text-muted-foreground"
   621→                        >
   622→                          {col.type}
   623→                        </Badge>
   624→                      </span>
   625→                    </li>
   626→                  ))}
   627→                </ul>
   628→              )}
   629→            </CardContent>
   630→          </Card>
   631→        </TabsContent>
   632→
   633→        {/* ── History ── */}
   634→        <TabsContent value="history" className="mt-3">
   635→          {/* R14: version compare (schema + row diff) */}
   636→          <div className="mb-4">
   637→            <DatasetDiffPanel
   638→              datasetId={id}
   639→              versions={detail.versions.map((v) => ({ versionNumber: v.versionNumber, rowCount: v.rowCount }))}
   640→            />
   641→          </div>
   642→
   643→          {/* Retention (M2): keepLastN select + manual prune */}
   644→          <div className="mb-4 flex flex-col gap-3 rounded-lg border border-border bg-muted/40 p-3 sm:flex-row sm:items-center sm:justify-between">
   645→            <div className="flex min-w-0 items-start gap-2.5">
   646→              <Archive className="mt-0.5 h-4 w-4 shrink-0 text-muted-foreground" aria-hidden />
   647→              <div className="min-w-0">
   648→                <p className="text-xs font-medium text-foreground">
   649→                  Version retention{' '}
   650→                  <span className="ml-1 font-mono text-[10px] text-muted-foreground">
   651→                    {keepLastN === 0
   652→                      ? 'unlimited'
   653→                      : `keep last ${keepLastN ?? 20} of ${detail.dataset.versionsCount}`}
   654→                  </span>
   655→                </p>
   656→                <p className="mt-0.5 text-[11px] leading-4 text-muted-foreground">
   657→                  Applies automatically on every new version (ingest, pipeline, transform,
   658→                  action). Pruning never touches the current version.
   659→                </p>
   660→              </div>
   661→            </div>
   662→            <div className="flex shrink-0 items-center gap-2">
   663→              <Select
   664→                value={String(retentionValue)}
   665→                onValueChange={(v) => retentionMutation.mutate(Number(v))}
   666→              >
   667→                <SelectTrigger
   668→                  aria-label="Retention: keep newest N versions"
   669→                  className="h-8 w-[150px] font-mono text-xs"
   670→                >
   671→                  <SelectValue />
   672→                </SelectTrigger>
   673→                <SelectContent>
   674→                  {RETENTION_CHOICES.map((n) => (
   675→                    <SelectItem key={n} value={String(n)} className="font-mono text-xs">
   676→                      {n === 0 ? 'Unlimited' : `Keep last ${n}`}
   677→                    </SelectItem>
   678→                  ))}
   679→                  {typeof retentionValue === 'number' &&
   680→                    !(RETENTION_CHOICES as readonly number[]).includes(retentionValue) && (
   681→                      <SelectItem value={String(retentionValue)} className="font-mono text-xs">
   682→                        Keep last {retentionValue} (custom)
   683→                      </SelectItem>
   684→                    )}
   685→                </SelectContent>
   686→              </Select>
   687→              <GatedButton
   688→                allowed={canEdit}
   689→                reason="requires editor role"
   690→                size="sm"
   691→                variant="outline"
   692→                onClick={() => pruneMutation.mutate()}
   693→                disabled={pruneMutation.isPending || retentionMutation.isPending}
   694→                className="h-8 text-xs"
   695→              >
   696→                {pruneMutation.isPending ? (
   697→                  <Loader2 className="h-3.5 w-3.5 animate-spin" aria-hidden />
   698→                ) : (
   699→                  <Archive className="h-3.5 w-3.5" aria-hidden />
   700→                )}
   701→                Prune now
   702→              </GatedButton>
   703→            </div>
   704→          </div>
   705→
   706→          <Card>
   707→            <CardContent className="p-4">
   708→              <ol className="relative ml-3 space-y-4 border-l border-border pl-6">
   709→                {detail.versions.map((v, idx) => {
   710→                  const isLatest = idx === 0
   711→                  const isSelected = v.versionNumber === version
   712→                  return (
   713→                    <li key={v.id} className="relative">
   714→                      <span
   715→                        aria-hidden
   716→                        className={cn(
   717→                          'absolute -left-[31px] top-0.5 h-3 w-3 rounded-full border-2',
   718→                          isLatest
   719→                            ? 'border-emerald-500 bg-emerald-500/40'
   720→                            : 'border-border bg-muted',
   721→                        )}
   722→                      />
   723→                      <div
   724→                        className={cn(
   725→                          'rounded-md border px-3 py-2 transition-colors',
   726→                          isSelected
   727→                            ? 'border-foreground/25 bg-accent/40'
   728→                            : 'border-border bg-background',
   729→                        )}
   730→                      >
   731→                        <div className="flex flex-wrap items-center justify-between gap-2">
   732→                          <button
   733→                            type="button"
   734→                            onClick={() => {
   735→                              setSelectedVersion(v.versionNumber)
   736→                              setPage(1)
   737→                            }}
   738→                            className="font-mono text-sm font-medium text-foreground hover:underline underline-offset-2"
   739→                            aria-label={`Show version ${v.versionNumber}`}
   740→                          >
   741→                            v{v.versionNumber}
   742→                          </button>
   743→                          <span className="font-mono text-xs text-emerald-600 dark:text-emerald-400">
   744→                            {v.rowCount.toLocaleString()} rows
   745→                          </span>
   746→                        </div>
   747→                        <div className="mt-1 flex flex-wrap items-center gap-x-3 gap-y-0.5 font-mono text-[11px] text-muted-foreground">
   748→                          <span title={v.createdBy ?? undefined}>by {shortId(v.createdBy, 10)}</span>
   749→                          <span title={format(parseISO(v.createdAt), 'MMM d, yyyy HH:mm:ss')}>
   750→                            {format(parseISO(v.createdAt), 'MMM d, HH:mm:ss')}
   751→                          </span>
   752→                          {isLatest && (
   753→                            <span className="rounded border border-emerald-500/40 bg-emerald-500/10 px-1 text-[10px] text-emerald-600 dark:text-emerald-400">
   754→                              latest
   755→                            </span>
   756→                          )}
   757→                          {/* P2: publication lifecycle + quality state chips */}
   758→                          {v.lifecycle === 'draft' && (
   759→                            <span className="rounded border border-amber-500/40 bg-amber-500/10 px-1 text-[10px] text-amber-600 dark:text-amber-400">
   760→                              draft
   761→                            </span>
   762→                          )}
   763→                          {v.versionNumber === detail.dataset.publishedVersion && v.lifecycle !== 'draft' && (
   764→                            <span className="rounded border border-border bg-muted px-1 text-[10px] text-foreground/70">
   765→                              published
   766→                            </span>
   767→                          )}
   768→                          {v.qualityState && v.qualityState !== 'not_evaluated' && (
   769→                            <span
   770→                              className={cn(
   771→                                'rounded border px-1 text-[10px]',
   772→                                v.qualityState === 'passed'
   773→                                  ? 'border-emerald-500/40 bg-emerald-500/10 text-emerald-600 dark:text-emerald-400'
   774→                                  : v.qualityState === 'warned'
   775→                                    ? 'border-amber-500/40 bg-amber-500/10 text-amber-600 dark:text-amber-400'
   776→                                    : 'border-rose-500/40 bg-rose-500/10 text-rose-600 dark:text-rose-400',
   777→                              )}
   778→                              title={`quality ${v.qualityState}`}
   779→                            >
   780→                              ● {v.qualityState}
   781→                            </span>
   782→                          )}
   783→                          {(v as { branch?: string }).branch && (v as { branch?: string }).branch !== 'main' && (
   784→                            <span className="rounded border border-border bg-muted px-1 text-[10px] text-muted-foreground">
   785→                              ⎇ {(v as { branch?: string }).branch}
   786→                            </span>
   787→                          )}
   788→                          {isSelected && !isLatest && (
   789→                            <span className="rounded border border-border bg-muted px-1 text-[10px] text-muted-foreground">
   790→                              viewing
   791→                            </span>
   792→                          )}
   793→                        </div>
   794→                      </div>
   795→                    </li>
   796→                  )
   797→                })}
   798→              </ol>
   799→            </CardContent>
   800→          </Card>
   801→        </TabsContent>
   802→
   803→        {/* ── Quality (R14) ── */}
   804→        <TabsContent value="quality" className="mt-3">
   805→          <DatasetQualityPanel
   806→            datasetId={id}
   807→            canEdit={canEdit}
   808→            columns={detail.current?.schema ?? []}
   809→            qualityGate={detail.dataset.qualityGate ?? false}
   810→            publishedVersion={detail.dataset.publishedVersion ?? null}
   811→          />
   812→        </TabsContent>
   813→
   814→        {/* ── Discussion (P4 collaboration) ── */}
   815→        <TabsContent value="discussion" className="mt-3">
   816→          <DatasetDiscussion
   817→            datasetId={id}
   818→            currentVersion={version}
   819→            versions={detail.versions.map((v) => v.versionNumber)}
   820→          />
   821→        </TabsContent>
   822→      </Tabs>
   823→    </div>
