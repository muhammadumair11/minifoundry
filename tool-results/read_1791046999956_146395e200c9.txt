   220→/** Mono path with {placeholders} tinted amber. */
   221→function PathText({ path, className }: { path: string; className?: string }) {
   222→  const parts = path.split(/(\{[^}]+\})/g)
   223→  return (
   224→    <span className={className}>
   225→      {parts.map((part, i) =>
   226→        part.startsWith('{') && part.endsWith('}') ? (
   227→          <span key={i} className="text-amber-300/90">
   228→            {part}
   229→          </span>
   230→        ) : (
   231→          <span key={i}>{part}</span>
   232→        ),
   233→      )}
   234→    </span>
   235→  )
   236→}

   247→/** Hand-rolled JSON syntax highlighter (no deps): keys zinc, strings emerald,
   248→    numbers amber, literals teal. Operates on the pretty-printed text. */
   249→function JsonHighlight({ text }: { text: string }) {
   250→  const nodes = useMemo(() => {
   251→    const out: React.ReactNode[] = []
   252→    const re = /("(?:\\u[0-9a-fA-F]{4}|\\[^u]|[^\\"])*")(\s*:)|("(?:\\u[0-9a-fA-F]{4}|\\[^u]|[^\\"])*")|(-?\d+(?:\.\d+)?(?:[eE][+-]?\d+)?)|\b(true|false|null)\b/g
   253→    let last = 0
   254→    let m: RegExpExecArray | null
   255→    let k = 0
   256→    while ((m = re.exec(text)) !== null) {
   257→      if (m.index > last) out.push(<span key={k++}>{text.slice(last, m.index)}</span>)
   258→      if (m[1] !== undefined) {
   259→        // Property key (followed by colon)
   260→        out.push(
   261→          <span key={k++} className="text-zinc-300">
   262→            {m[1]}
   263→          </span>,
   264→          <span key={k++}>{m[2]}</span>,
   265→        )
   266→      } else if (m[3] !== undefined) {
   267→        out.push(
   268→          <span key={k++} className="text-emerald-300">
   269→            {m[3]}
   270→          </span>,
   271→        )
   272→      } else if (m[4] !== undefined) {
   273→        out.push(
   274→          <span key={k++} className="text-amber-300">
   275→            {m[4]}
   276→          </span>,
   277→        )
   278→      } else if (m[5] !== undefined) {
   279→        out.push(
   280→          <span key={k++} className="text-teal-300">
   281→            {m[5]}
   282→          </span>,
   283→        )
   284→      }
   285→      last = re.lastIndex
   286→    }
   287→    if (last < text.length) out.push(<span key={k++}>{text.slice(last)}</span>)
   288→    return out
   289→  }, [text])
   290→  return <pre className="font-mono text-xs leading-5 text-zinc-400">{nodes}</pre>
   291→}

   341→export default function ApiDocsPage() {
   342→  const [selectedId, setSelectedId] = useState<string>('whoami')
   343→  const endpoint = ENDPOINTS.find((e) => e.id === selectedId) ?? ENDPOINTS[0]
   344→
   345→  // Param values, keyed per endpoint so switching tabs preserves entries.
   346→  const [paramValues, setParamValues] = useState<Record<string, string>>({})
   347→  const getParam = useCallback(
   348→    (epId: string, name: string) => paramValues[`${epId}:${name}`] ?? '',
   349→    [paramValues],
   350→  )
   351→  const setParam = useCallback((epId: string, name: string, value: string) => {
   352→    setParamValues((prev) => ({ ...prev, [`${epId}:${name}`]: value }))
   353→  }, [])
   354→
   355→  // Token state: saved secrets (localStorage) + free-text secret input.
   356→  const [savedTokens, setSavedTokens] = useState<SavedApiToken[]>([])
   357→  const [tokenInput, setTokenInput] = useState('')
   358→  const [selectedSavedId, setSelectedSavedId] = useState<string>('none')
   359→  const [showToken, setShowToken] = useState(false)
   360→
   361→  useEffect(() => {
   362→    setSavedTokens(loadSavedTokens())
   363→  }, [])
   364→
   365→  const applySaved = (id: string) => {
   366→    setSelectedSavedId(id)
   367→    if (id === 'none') return
   368→    const found = savedTokens.find((t) => t.id === id)
   369→    if (found) setTokenInput(found.secret)
   370→  }
   371→
   372→  const handleClearSaved = () => {
   373→    clearSavedTokens()
   374→    setSavedTokens([])
   375→    setSelectedSavedId('none')
   376→    toast.success('Saved secrets cleared', {
   377→      description: `localStorage key ${SAVED_TOKENS_KEY} is now empty.`,
   378→    })
   379→  }
   380→
   381→  const trimmedToken = tokenInput.trim()
   382→  const tokenLooksValid = trimmedToken.startsWith('mf_live_')
   383→
   384→  // Which path param is focused → drives smart suggestions.
   385→  const [focusKey, setFocusKey] = useState<string | null>(null)
   386→  const focusedParam = endpoint.params.find((p) => `${endpoint.id}:${p.name}` === focusKey) ?? null
   387→
   388→  // ── Smart suggestions (TanStack Query, fetched only when focused) ──
   389→  const v1Headers = useMemo(
   390→    () => (trimmedToken ? { Authorization: `Bearer ${trimmedToken}` } : undefined),
   391→    [trimmedToken],
   392→  )
   393→
   394→  const datasetSuggestQuery = useQuery({
   395→    queryKey: ['v1-suggest-datasets', trimmedToken.slice(0, 12)],
   396→    queryFn: async () => {
   397→      const res = await fetch(`${BASE_PATH}/datasets?pageSize=1`, { headers: v1Headers })
   398→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
   399→      return (await res.json()) as { data: { id: string; name: string }[] }
   400→    },
   401→    enabled: focusedParam?.suggest === 'datasetId' && !!trimmedToken,
   402→    staleTime: Infinity,
   403→    retry: false,
   404→  })
   405→
   406→  const pipelineSuggestQuery = useQuery({
   407→    queryKey: ['v1-suggest-pipelines', trimmedToken.slice(0, 12)],
   408→    queryFn: async () => {
   409→      const res = await fetch(`${BASE_PATH}/pipelines?pageSize=1`, { headers: v1Headers })
   410→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
   411→      return (await res.json()) as { data: { id: string; name: string }[] }
   412→    },
   413→    enabled: focusedParam?.suggest === 'pipelineId' && !!trimmedToken,
   414→    staleTime: Infinity,
   415→    retry: false,
   416→  })
   417→
   418→  const ontologySuggestQuery = useQuery<OntologyListResponse>({
   419→    queryKey: ['ontology'],
   420→    queryFn: () => apiJson<OntologyListResponse>('/api/ontology'),
   421→    enabled: focusedParam?.suggest === 'objectType',
   422→    staleTime: Infinity,
   423→  })
   424→
   425→  const currentType = getParam(endpoint.id, 'type')
   426→  const pkSuggestQuery = useQuery({
   427→    queryKey: ['v1-suggest-objects', currentType, trimmedToken.slice(0, 12)],
   428→    queryFn: async () => {
   429→      const res = await fetch(`${BASE_PATH}/objects/${encodeURIComponent(currentType)}?pageSize=1`, {
   430→        headers: v1Headers,
   431→      })
   432→      if (!res.ok) throw new Error(`HTTP ${res.status}`)
   433→      return (await res.json()) as { data: { pk: unknown }[] }
   434→    },
   435→    enabled: focusedParam?.suggest === 'objectPk' && !!currentType && !!trimmedToken,
   436→    staleTime: Infinity,
   437→    retry: false,
   438→  })
   439→
   440→  // Auto-fill a focused, still-empty input once its suggestion arrives.
   441→  const autoFilledRef = useRef<Set<string>>(new Set())
   442→  useEffect(() => {
   443→    if (!focusedParam) return
   444→    const key = `${endpoint.id}:${focusedParam.name}`
   445→    if (getParam(endpoint.id, focusedParam.name)) return
   446→    if (autoFilledRef.current.has(key)) return
   447→    let suggestion: string | null = null
   448→    if (focusedParam.suggest === 'datasetId')
   449→      suggestion = datasetSuggestQuery.data?.data[0]?.id ?? null
   450→    else if (focusedParam.suggest === 'pipelineId')
   451→      suggestion = pipelineSuggestQuery.data?.data[0]?.id ?? null
   452→    else if (focusedParam.suggest === 'objectType')
   453→      suggestion = ontologySuggestQuery.data?.objectTypes[0]?.apiName ?? null
   454→    else if (focusedParam.suggest === 'objectPk') {
   455→      const pk = pkSuggestQuery.data?.data[0]?.pk
   456→      suggestion = pk === null || pk === undefined ? null : String(pk)
   457→    }
   458→    if (suggestion) {
   459→      autoFilledRef.current.add(key)
   460→      setParam(endpoint.id, focusedParam.name, suggestion)
   461→    }
   462→  }, [
   463→    focusedParam,
   464→    endpoint.id,
   465→    getParam,
   466→    setParam,
   467→    datasetSuggestQuery.data,
   468→    pipelineSuggestQuery.data,
   469→    ontologySuggestQuery.data,
   470→    pkSuggestQuery.data,
   471→  ])
   472→
   473→  // ── Resolved path + query string ────────────────────────────────────
   474→  const resolvedPath = useMemo(() => {
   475→    let path = endpoint.pathTemplate
   476→    for (const p of endpoint.params) {
   477→      if (p.where === 'path') {
   478→        path = path.replace(`{${p.name}}`, encodeURIComponent(getParam(endpoint.id, p.name)) || `{${p.name}}`)
   479→      }
   480→    }
   481→    const qs = new URLSearchParams()
   482→    for (const p of endpoint.params) {
   483→      if (p.where === 'query') {
   484→        const v = getParam(endpoint.id, p.name).trim()
   485→        if (v) qs.set(p.name, v)
   486→      }
   487→    }
   488→    const query = qs.toString()
   489→    return query ? `${path}?${query}` : path
   490→  }, [endpoint, getParam])
   491→
   492→  const curlCommand = `curl -H "Authorization: Bearer $TOKEN" "${ORIGIN}${BASE_PATH}${resolvedPath}"`
   493→
   494→  // ── Try-it send ─────────────────────────────────────────────────────
   495→  const [pending, setPending] = useState(false)
   496→  const [result, setResult] = useState<TryResult | null>(null)
   497→  const resultRef = useRef<HTMLDivElement | null>(null)
   498→
   499→  const handleSend = async () => {
   500→    if (!trimmedToken || pending) return
   501→    setPending(true)
   502→    const started = performance.now()
   503→    try {
   504→      const res = await fetch(`${BASE_PATH}${resolvedPath}`, {
   505→        headers: { Authorization: `Bearer ${trimmedToken}` },
   506→      })
   507→      const bodyText = await res.text()
   508→      const headers: Record<string, string> = {}
   509→      for (const key of [...RATE_HEADER_KEYS, 'retry-after', 'content-type', 'content-disposition']) {
   510→        const v = res.headers.get(key)
   511→        if (v) headers[key] = v
   512→      }
   513→      setResult({
   514→        status: res.status,
   515→        statusText: res.statusText,
   516→        durationMs: Math.round(performance.now() - started),
   517→        sizeBytes: new TextEncoder().encode(bodyText).length,
   518→        headers,
   519→        bodyText,
   520→        isJson: (res.headers.get('content-type') ?? '').includes('json'),
   521→        requestPath: resolvedPath,
   522→      })
   523→    } catch {
   524→      toast.error('Request failed', {
   525→        description: 'Network error — is the dev server running?',
   526→      })
   527→    } finally {
   528→      setPending(false)
   529→      // Let the panel paint, then scroll it into view on small screens.
   530→      requestAnimationFrame(() =>
   531→        resultRef.current?.scrollIntoView({ behavior: 'smooth', block: 'nearest' }),
   532→      )
   533→    }
   534→  }
   535→
   536→  const handleCopyCurl = async () => {
   537→    try {
   538→      await navigator.clipboard.writeText(curlCommand)
   539→      toast.success('curl command copied', {
   540→        description: 'Paste it in a shell and export TOKEN=mf_live_… first.',
