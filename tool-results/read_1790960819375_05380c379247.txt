    53→function useDebouncedValue<T>(value: T, delayMs: number): T {
    54→  const [debounced, setDebounced] = useState(value)
    55→  useEffect(() => {
    56→    const t = setTimeout(() => setDebounced(value), delayMs)
    57→    return () => clearTimeout(t)
    58→  }, [value, delayMs])
    59→  return debounced
    60→}

    62→/** Compact value renderer for object cells / property rows. */
    63→function renderValue(value: unknown): React.ReactNode {
    64→  if (value === null || value === undefined || value === '') {
    65→    return <span className="text-zinc-700">∅</span>
    66→  }
    67→  if (typeof value === 'boolean') {
    68→    return <span className={value ? 'text-amber-400' : 'text-zinc-500'}>{String(value)}</span>
    69→  }
    70→  if (typeof value === 'number') {
    71→    return <span className="tabular-nums">{value.toLocaleString()}</span>
    72→  }
    73→  return String(value)
    74→}

    78→function ActiveActionDialog({
    79→  actionApiName,
    80→  prefillPk,
    81→  onClose,
    82→}: {
    83→  actionApiName: string
    84→  prefillPk: string | null
    85→  onClose: () => void
    86→}) {
    87→  const actionQuery = useQuery<ActionResponse>({
    88→    queryKey: ['action', actionApiName],
    89→    queryFn: () => apiJson<ActionResponse>(`/api/actions/${encodeURIComponent(actionApiName)}`),
    90→    enabled: actionApiName !== '',
    91→  })
    92→
    93→  useEffect(() => {
    94→    if (actionQuery.isError) {
    95→      toast.error('Could not load action', {
    96→        description:
    97→          actionQuery.error instanceof Error ? actionQuery.error.message : 'Unknown error.',
    98→      })
    99→      onClose()
   100→    }
   101→  }, [actionQuery.isError, actionQuery.error, onClose])
   102→
   103→  if (!actionQuery.data) return null
   104→
   105→  return (
   106→    <ActionDialog
   107→      action={actionQuery.data.action}
   108→      open
   109→      onOpenChange={(next) => {
   110→        if (!next) onClose()
   111→      }}
   112→      prefillPk={prefillPk}
   113→    />
   114→  )
   115→}

   313→// ─── Workshop view ────────────────────────────────────────────────────

   315→function WorkshopView() {
   316→  const routeParams = useParams<{ type: string }>()
   317→  const typeApiName = routeParams?.type ?? ''
   318→  const searchParams = useSearchParams()
   319→  const actionParam = searchParams.get('action')
   320→  const pkParam = searchParams.get('pk')
   321→
   322→  const [searchInput, setSearchInput] = useState('')
   323→  const [page, setPage] = useState(1)
   324→  const [selectedPk, setSelectedPk] = useState<string | null>(null)
   325→  const [activeActionApiName, setActiveActionApiName] = useState<string | null>(null)
   326→
   327→  const debouncedSearch = useDebouncedValue(searchInput, 400)
   328→
   329→  // ?pk= → preselect (and type change → reset) by syncing during render.
   330→  const [pkSync, setPkSync] = useState('')
   331→  const pkSignal = `${typeApiName}\u0000${pkParam ?? ''}`
   332→  if (pkSync !== pkSignal) {
   333→    setPkSync(pkSignal)
   334→    setSelectedPk(pkParam)
   335→  }
   336→
   337→  // ?action= → auto-open the action dialog once per param value (closing it
   338→  // does not re-open while the URL keeps the same ?action=).
   339→  const [consumedActionParam, setConsumedActionParam] = useState<string | null>(null)
   340→  if (actionParam && actionParam !== consumedActionParam) {
   341→    setConsumedActionParam(actionParam)
   342→    setActiveActionApiName(actionParam)
   343→  }
   344→
   345→  // Ontology list — type meta + enabled actions.
   346→  const ontologyQuery = useQuery<OntologyListResponse>({
   347→    queryKey: ['ontology'],
   348→    queryFn: () => apiJson<OntologyListResponse>('/api/ontology'),
   349→  })
   350→  const typeInfo = useMemo(
   351→    () => ontologyQuery.data?.objectTypes.find((t) => t.apiName === typeApiName) ?? null,
   352→    [ontologyQuery.data, typeApiName],
   353→  )
   354→  const enabledActions = useMemo(
   355→    () => (typeInfo?.actions ?? []).filter((a) => a.enabled),
   356→    [typeInfo],
   357→  )
   358→
   359→  // Object table.
   360→  const objectsQuery = useQuery<ObjectsListResponse>({
   361→    queryKey: ['objects', typeApiName, debouncedSearch, page],
   362→    queryFn: () =>
   363→      apiJson<ObjectsListResponse>(
   364→        `/api/objects?type=${encodeURIComponent(typeApiName)}&query=${encodeURIComponent(debouncedSearch)}&page=${page}&pageSize=${PAGE_SIZE}`,
   365→      ),
   366→    enabled: typeApiName !== '',
   367→    placeholderData: (prev) => prev,
   368→  })
   369→  const rows = objectsQuery.data?.rows ?? []
   370→  const allColumns = objectsQuery.data?.columns ?? []
   371→  const pkColumn = objectsQuery.data?.objectType.primaryKey ?? null
   372→  // The table already renders a dedicated pk column — drop the property that mirrors it.
   373→  const tableColumns = useMemo(
   374→    () => allColumns.filter((c) => !c.hideInTable && c.apiName !== pkColumn),
   375→    [allColumns, pkColumn],
   376→  )
   377→  const total = objectsQuery.data?.total ?? 0
   378→  const totalPages = Math.max(1, Math.ceil(total / (objectsQuery.data?.pageSize ?? PAGE_SIZE)))
   379→  const titleProperty = objectsQuery.data?.objectType.titleProperty ?? null
   380→
   381→  const unknownType = !ontologyQuery.isLoading && !!ontologyQuery.data && !typeInfo
   382→
   383→  return (
   384→    <div className="mx-auto w-full max-w-6xl px-4 py-6 sm:py-8">
   385→      {/* a) Type header */}
   386→      <header className="mb-5">
   387→        {ontologyQuery.isLoading ? (
   388→          <div className="flex items-center gap-3">
   389→            <Skeleton className="h-10 w-10 bg-zinc-800" />
   390→            <div className="space-y-1.5">
   391→              <Skeleton className="h-6 w-40 bg-zinc-800" />
   392→              <Skeleton className="h-3.5 w-56 bg-zinc-800" />
   393→            </div>
   394→          </div>
   395→        ) : typeInfo ? (
   396→          <div className="flex flex-col gap-3 sm:flex-row sm:items-start sm:justify-between">
   397→            <div className="flex items-start gap-3">
   398→              <TypeIconBadge type={typeInfo} />
   399→              <div>
   400→                <div className="flex flex-wrap items-baseline gap-2">
   401→                  <h1 className="text-xl font-semibold tracking-tight text-zinc-100 sm:text-2xl">
   402→                    {typeInfo.displayName}
   403→                  </h1>
   404→                  <span className="font-mono text-xs text-zinc-500">{typeInfo.apiName}</span>
   405→                </div>
   406→                <p className="mt-1 flex flex-wrap items-center gap-x-3 gap-y-1 font-mono text-[11px] text-zinc-500">
   407→                  <span>
   408→                    {typeInfo.objectCount ?? '—'} <span className="text-zinc-600">objects</span>
   409→                  </span>
   410→                  {typeInfo.sourceDatasetId && (

   538→                <div className="p-6 text-center text-sm text-zinc-500">
   539→                  {debouncedSearch
   540→                    ? `No objects match “${debouncedSearch}”.`
   541→                    : 'No objects in the backing dataset yet.'}
   542→                </div>
   543→              ) : (
   544→                <div className="foundry-scroll max-h-[520px] overflow-auto">
   545→                  <Table>
   546→                    <TableHeader className="sticky top-0 z-10 bg-zinc-900">
   547→                      <TableRow className="border-zinc-800 hover:bg-transparent">
   548→                        <TableHead className="h-8 px-4 font-mono text-[10px] uppercase tracking-wider text-zinc-500">
   549→                          {objectsQuery.data?.objectType.primaryKey ?? 'pk'}
   550→                        </TableHead>
   551→                        {tableColumns.map((c) => (
   552→                          <TableHead
   553→                            key={c.apiName}
   554→                            className="h-8 whitespace-nowrap px-3 font-mono text-[10px] uppercase tracking-wider text-zinc-500"
   555→                          >
   556→                            {c.displayName}
   557→                          </TableHead>
   558→                        ))}
   559→                      </TableRow>
   560→                    </TableHeader>
   561→                    <TableBody>
   562→                      {rows.map((row) => {
   563→                        const selected = row.pk === selectedPk
   564→                        return (
   565→                          <TableRow
   566→                            key={row.pk}
   567→                            aria-selected={selected}
   568→                            onClick={() => setSelectedPk(row.pk)}
   569→                            className={cn(
   570→                              'cursor-pointer border-zinc-800/70',
   571→                              selected
   572→                                ? 'border-l-2 border-l-emerald-400 bg-zinc-800/40 hover:bg-zinc-800/40'
   573→                                : 'hover:bg-zinc-800/30',
   574→                            )}
   575→                          >
   576→                            <TableCell className="px-4 py-2 font-mono text-xs text-zinc-500">
   577→                              {row.pk}
   578→                            </TableCell>
   579→                            {tableColumns.map((c) => (
   580→                              <TableCell key={c.apiName} className="max-w-[220px] px-3 py-2">
   581→                                {c.isTitle ? (
   582→                                  <button
   583→                                    type="button"
   584→                                    onClick={(e) => {
   585→                                      e.stopPropagation()
   586→                                      setSelectedPk(row.pk)
   587→                                    }}
   588→                                    className="max-w-full truncate text-left text-xs font-semibold text-emerald-400 hover:underline"
   589→                                    title={`Select ${row.pk}`}
   590→                                  >
   591→                                    {renderValue(row.values[c.apiName])}
   592→                                  </button>
   593→                                ) : (
   594→                                  <span className="block max-w-[220px] truncate font-mono text-xs text-zinc-300">
   595→                                    {renderValue(row.values[c.apiName])}
   596→                                  </span>
   597→                                )}
   598→                              </TableCell>
   599→                            ))}
   600→                          </TableRow>
   601→                        )
   602→                      })}
   603→                    </TableBody>
   604→                  </Table>
   605→                </div>
   606→              )}
   607→
   608→              {/* Pagination footer */}
   609→              {!objectsQuery.isLoading && rows.length > 0 && (
   610→                <div className="flex items-center justify-between gap-2 border-t border-zinc-800 px-4 py-2.5">
   611→                  <p className="font-mono text-[11px] text-zinc-500">
   612→                    {total} objects · page {page} / {totalPages}
   613→                  </p>
   614→                  <div className="flex items-center gap-1.5">
   615→                    <Button
   616→                      size="sm"
   617→                      variant="outline"
   618→                      className="h-7 border-zinc-800 bg-zinc-950/60 px-2 text-zinc-300 hover:bg-zinc-800 hover:text-zinc-100"
   619→                      disabled={page <= 1}
   620→                      onClick={() => setPage((p) => Math.max(1, p - 1))}
   621→                      aria-label="Previous page"
   622→                    >
   623→                      <ChevronLeft className="h-3.5 w-3.5" aria-hidden />
   624→                    </Button>
   625→                    <Button
   626→                      size="sm"
   627→                      variant="outline"
   628→                      className="h-7 border-zinc-800 bg-zinc-950/60 px-2 text-zinc-300 hover:bg-zinc-800 hover:text-zinc-100"
   629→                      disabled={page >= totalPages}
   630→                      onClick={() => setPage((p) => Math.min(totalPages, p + 1))}
   631→                      aria-label="Next page"
   632→                    >
   633→                      <ChevronRight className="h-3.5 w-3.5" aria-hidden />
   634→                    </Button>
   635→                  </div>
   636→                </div>
   637→              )}
   638→            </CardContent>
   639→          </Card>
   640→
   641→          {/* c) RIGHT: detail panel */}
   642→          <div className="w-full shrink-0 lg:sticky lg:top-[72px] lg:w-96">
   643→            <Card className="border-zinc-800 bg-zinc-900/60">
   644→              {selectedPk ? (
   645→                <ObjectDetailPanel
   646→                  typeApiName={typeApiName}
   647→                  selectedPk={selectedPk}
   648→                  columns={allColumns}
   649→                  titleProperty={titleProperty}
   650→                  onClear={() => setSelectedPk(null)}
   651→                />
   652→              ) : (
   653→                <CardContent className="flex flex-col items-center gap-2 p-8 text-center">
   654→                  <Compass className="h-7 w-7 text-zinc-600" aria-hidden />
   655→                  <p className="text-sm font-medium text-zinc-400">No object selected</p>
   656→                  <p className="text-xs text-zinc-600">
   657→                    Click a row to inspect its properties and links.
   658→                  </p>
   659→                </CardContent>
   660→              )}
   661→            </Card>
   662→          </div>
   663→        </div>
   664→      )}
   665→
   666→      {/* d) Action dialog (dropdown picks + ?action= deep link) */}
   667→      {activeActionApiName && (
   668→        <ActiveActionDialog
   669→          actionApiName={activeActionApiName}
   670→          prefillPk={selectedPk}
   671→          onClose={() => setActiveActionApiName(null)}
   672→        />
   673→      )}
   674→    </div>
   675→  )
   676→}
