   995→export function shortId(id: string | null | undefined, len = 8): string {
   996→  if (!id) return '—'
   997→  return id.length <= len ? id : id.slice(0, len)
   998→}

  1000→export function formatDuration(startedAt: string, finishedAt: string | null): string {
  1001→  if (!finishedAt) return '—'
  1002→  const ms = new Date(finishedAt).getTime() - new Date(startedAt).getTime()
  1003→  if (ms < 1000) return `${ms}ms`
  1004→  if (ms < 60_000) return `${(ms / 1000).toFixed(1)}s`
  1005→  const m = Math.floor(ms / 60_000)
  1006→  const s = Math.round((ms % 60_000) / 1000)
  1007→  return `${m}m ${s}s`
  1008→}
  1009→
  1010→/** Strict relative time ("3m ago") for data-UI timestamps; '—' for null/invalid. */
  1011→export function formatRelative(iso: string | null | undefined): string {
  1012→  if (!iso) return '—'
  1013→  const d = new Date(iso)
  1014→  if (Number.isNaN(d.getTime())) return '—'
  1015→  return formatDistanceToNowStrict(d, { addSuffix: true })
  1016→}
  1017→
  1018→/** "every 10m" / "every 1h" / "every 1d" — human label for schedule intervals. */
  1019→export function formatInterval(minutes: number): string {
  1020→  if (minutes >= 1440 && minutes % 1440 === 0) return `every ${minutes / 1440}d`
  1021→  if (minutes >= 60 && minutes % 60 === 0) return `every ${minutes / 60}h`
  1022→  return `every ${minutes}m`
  1023→}

  1025→// Data-freshness thresholds (Task 2-c spec): fresh < 15m, aging < 2h, else stale.
  1026→export const FRESH_THRESHOLD_MINUTES = 15
  1027→export const STALE_THRESHOLD_MINUTES = 120
  1028→
  1029→export type FreshnessLevel = 'fresh' | 'aging' | 'stale'
  1030→
  1031→export function freshnessLevel(freshnessMinutes: number): FreshnessLevel {
  1032→  if (freshnessMinutes < FRESH_THRESHOLD_MINUTES) return 'fresh'
  1033→  if (freshnessMinutes < STALE_THRESHOLD_MINUTES) return 'aging'
  1034→  return 'stale'
  1035→}
