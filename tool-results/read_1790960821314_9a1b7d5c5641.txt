    53→    53→function useDebouncedValue<T>(value: T, delayMs: number): T {
    54→    54→  const [debounced, setDebounced] = useState(value)
    55→    55→  useEffect(() => {
    56→    56→    const t = setTimeout(() => setDebounced(value), delayMs)
    57→    57→    return () => clearTimeout(t)
    58→    58→  }, [value, delayMs])
    59→    59→  return debounced
    60→    60→}

    62→    62→/** Compact value renderer for object cells / property rows. */
    63→    63→function renderValue(value: unknown): React.ReactNode {
    64→    64→  if (value === null || value === undefined || value === '') {
    65→    65→    return <span className="text-zinc-700">∅</span>
    66→    66→  }
    67→    67→  if (typeof value === 'boolean') {
    68→    68→    return <span className={value ? 'text-amber-400' : 'text-zinc-500'}>{String(value)}</span>
    69→    69→  }
    70→    70→  if (typeof value === 'number') {
    71→    71→    return <span className="tabular-nums">{value.toLocaleString()}</span>
    72→    72→  }
    73→    73→  return String(value)
    74→    74→}
