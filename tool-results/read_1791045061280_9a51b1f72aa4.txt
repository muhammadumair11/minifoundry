   158→function NavLink({
   159→  item,
   160→  rail,
   161→  onNavigate,
   162→}: {
   163→  item: NavItem
   164→  rail: boolean
   165→  onNavigate?: () => void
   166→}) {
   167→  const pathname = usePathname()
   168→  const active = isActive(pathname, item.href)
   169→  const link = (
   170→    <Link
   171→      href={item.href}
   172→      onClick={onNavigate}
   173→      aria-current={active ? 'page' : undefined}
   174→      aria-label={rail ? item.label : undefined}
   175→      className={cn(
   176→        'group relative flex h-8 shrink-0 items-center rounded-md text-[13px] font-medium transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/50',
   177→        rail ? 'w-full justify-center px-0' : 'gap-2.5 px-2.5',
   178→        active
   179→          ? 'bg-emerald-500/[0.12] text-emerald-300'
   180→          : 'text-zinc-400 hover:bg-zinc-900 hover:text-zinc-100',
   181→      )}
   182→    >
   183→      {active && !rail && (
   184→        <span
   185→          className="absolute inset-y-[7px] left-0 w-[2.5px] rounded-full bg-emerald-400"
   186→          aria-hidden
   187→        />
   188→      )}
   189→      <item.icon
   190→        className={cn(
   191→          'size-4 shrink-0 transition-colors',
   192→          active
   193→            ? 'text-emerald-400'
   194→            : 'text-zinc-500 group-hover:text-zinc-300',
   195→        )}
   196→        aria-hidden
   197→      />
   198→      {!rail && <span className="truncate">{item.label}</span>}
   199→    </Link>
   200→  )
   201→
   202→  // Icon-only rail gets a hover tooltip; full sidebar labels speak for themselves.
   203→  if (!rail) return link
   204→  return (
   205→    <Tooltip>
   206→      <TooltipTrigger asChild>{link}</TooltipTrigger>
   207→      <TooltipContent
   208→        side="right"
   209→        sideOffset={10}
   210→        className="border-zinc-800 bg-zinc-900 px-2 py-1 text-xs text-zinc-200"
   211→      >
   212→        {item.label}
   213→      </TooltipContent>
   214→    </Tooltip>
   215→  )
   216→}

   402→function DemoAuthControls({ session }: { session: SessionResponse | undefined }) {

   482→/**
   483→ * Signed-in identity — avatar menu (task 7-c). Shows the real identity
   484→ * (name/email/role badge), a "Switch role (admin)" impersonation submenu for
   485→ * session admins, and Sign out (full cache invalidation so role-gated data
   486→ * downgrades back to demo level).
   487→ */
   488→function UserMenu({ session }: { session: SessionResponse }) {
   489→  const queryClient = useQueryClient()
   490→  const user: SessionUserInfo | null = session.session ?? null
   491→  // EFFECTIVE role (impersonation-aware); `user.role` stays the TRUE role.
   492→  const effectiveRole = user?.effectiveRole ?? session.role
   493→  const isAdminUser = user?.role === 'admin'
   494→  const impersonating = Boolean(user?.impersonating)
   495→
   496→  const logoutMutation = useMutation({
   497→    mutationFn: () => apiSend<{ ok: true }>('/api/auth/logout', 'POST'),
   498→    onSuccess: async () => {
   499→      // Demo-safe default: drop the simulator cookie to viewer so the
   500→      // anonymous session starts least-privileged (the bare no-cookie
   501→      // machine default would be admin). Best effort — ignore failures.
   502→      try {
   503→        await apiSend('/api/session/role', 'POST', { role: 'viewer' })
   504→      } catch {
   505→        /* demo default only — the refetch below settles the UI either way */
   506→      }
   507→      toast.success('Signed out', {
   508→        description: 'Demo mode — acting as viewer. Sign in to use a real identity.',
   509→      })
   510→      queryClient.invalidateQueries()
   511→    },
   512→    onError: (err) => toastApiError(err, 'Could not sign out'),
   513→  })
   514→
   515→  // Impersonation is REAL (session-carried): the server re-issues the session
   516→  // token with the effective role, so refetching ['session'] brings the truth
   517→  // back and every role-gated query re-resolves against the new role.
   518→  const switchRoleMutation = useMutation({
   519→    mutationFn: (next: FoundryRole) =>
   520→      apiSend<{ role: FoundryRole }>('/api/session/role', 'POST', { role: next }),
   521→    onSuccess: (res) => {
   522→      toast.success(
   523→        res.role === 'admin' ? 'Restored admin role' : `Acting as ${res.role}`,
   524→        {
   525→          description:
   526→            res.role === 'admin'
   527→              ? 'Full admin permissions are back.'
   528→              : 'API permissions now follow this role until you switch back or sign out.',
   529→        },
   530→      )
   531→      queryClient.invalidateQueries()
   532→    },
   533→    onError: (err) => toastApiError(err, 'Could not switch role'),
   534→  })
   535→
   536→  if (!user) return null
   537→  const initials = user.name
   538→    .split(/\s+/)
   539→    .filter(Boolean)
   540→    .map((w) => w[0])
   541→    .slice(0, 2)
   542→    .join('')
   543→    .toUpperCase()
   544→
   545→  return (
   546→    <DropdownMenu>
   547→      <DropdownMenuTrigger asChild>
   548→        <button
   549→          className="flex h-8 shrink-0 items-center gap-1.5 rounded-full border border-zinc-800 bg-zinc-900/60 py-1 pl-1 pr-1.5 transition-colors hover:border-zinc-700 hover:bg-zinc-800 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/50"
   550→          aria-label={`Account menu — ${user.name} (${effectiveRole}${impersonating ? ', impersonated' : ''})`}
   551→        >
   552→          <span
   553→            className="flex size-6 items-center justify-center rounded-full border border-emerald-500/40 bg-emerald-500/15 font-mono text-[9px] font-bold text-emerald-300"
   554→            aria-hidden
   555→          >
   556→            {initials}
   557→          </span>

   639→const SIDEBAR_KEY = 'mf.sidebar.collapsed'
   640→
   641→/**
   642→ * Rail/expanded preference, persisted in localStorage. Backed by
   643→ * useSyncExternalStore so the client picks up the stored value during
   644→ * hydration (no effect, no flash) and cross-tab updates stay in sync.
   645→ */
   646→function useSidebarCollapsed(): [boolean, (next: boolean) => void] {
   647→  const subscribe = useCallback((onStoreChange: () => void) => {
   648→    window.addEventListener('mf:sidebar', onStoreChange)
   649→    window.addEventListener('storage', onStoreChange)
   650→    return () => {
   651→      window.removeEventListener('mf:sidebar', onStoreChange)
   652→      window.removeEventListener('storage', onStoreChange)
   653→    }
   654→  }, [])
   655→  const collapsed = useSyncExternalStore(
   656→    subscribe,
   657→    () => window.localStorage.getItem(SIDEBAR_KEY) === '1',
   658→    () => false, // server snapshot: expanded
   659→  )
   660→  const setCollapsed = useCallback((next: boolean) => {
   661→    try {
   662→      window.localStorage.setItem(SIDEBAR_KEY, next ? '1' : '0')
   663→    } catch {
   664→      /* private mode etc. — preference just won't persist */
   665→    }
   666→    window.dispatchEvent(new Event('mf:sidebar'))
   667→  }, [])
   668→  return [collapsed, setCollapsed]
   669→}
