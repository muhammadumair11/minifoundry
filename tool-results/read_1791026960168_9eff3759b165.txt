   490→function WorkbookEditor({
   491→  transform,
   492→  detail,
   493→  detailLoading,
   494→  detailError,
   495→  datasets,
   496→  canEdit,
   497→  onDeleted,
   498→}: {
   499→  transform: TransformSummary
   500→  detail: TransformDetailResponse | null
   501→  detailLoading: boolean
   502→  detailError: string | null
   503→  datasets: DatasetSummary[]
   504→  canEdit: boolean
   505→  onDeleted: () => void
   506→}) {
   507→  const queryClient = useQueryClient()
   508→  const [draftCode, setDraftCode] = useState(transform.sourceCode)
   509→  const [renameOpen, setRenameOpen] = useState(false)
   510→  const [renameValue, setRenameValue] = useState(transform.name)
   511→  const [deleteOpen, setDeleteOpen] = useState(false)
   512→  const [activeTab, setActiveTab] = useState<'preview' | 'logs' | 'history'>('preview')
   513→  const [result, setResult] = useState<TransformRunResponse | null>(null)
   514→  const [resultsOpen, setResultsOpen] = useState(true)
   515→  const textareaRef = useRef<HTMLTextAreaElement>(null)
   516→  const gutterRef = useRef<HTMLDivElement>(null)
   517→  const [, setFocus] = useFocusMode()
   518→
   519→  // Keep the line-number gutter locked to the textarea's scroll position.
   520→  const syncGutter = () => {
   521→    if (gutterRef.current && textareaRef.current) {
   522→      gutterRef.current.scrollTop = textareaRef.current.scrollTop
   523→    }
   524→  }
   525→
   526→  const dirty = draftCode !== transform.sourceCode
   527→
   528→  const runsQuery = useQuery<TransformRunsListResponse>({
   529→    queryKey: ['transform-runs', transform.id],
   530→    queryFn: () =>
   531→      apiJson<TransformRunsListResponse>(`/api/transforms/${transform.id}/runs?limit=20`),
   532→  })
   533→
   534→  // The run shown in Preview/Logs: the fresh in-memory run, else the latest
   535→  // run from the detail payload.
   536→  const displayedRun: TransformRunSummary | null =
   537→    result?.run ?? detail?.runs[0] ?? null
   538→  const hasPreviewRows = !!result && result.preview.length >= 0 && result.run.status === 'success'
   539→

  1145→function PreviewTable({
  1146→  result,
  1147→  outputName,
  1148→}: {
  1149→  result: TransformRunResponse
  1150→  outputName: string
  1151→}) {
  1152→  const columns = result.schema
  1153→  const rows = result.preview
  1154→  const typeByName = useMemo(() => {
  1155→    const m = new Map<string, string>()
  1156→    for (const c of columns) m.set(c.name, c.type)
  1157→    return m
  1158→  }, [columns])
  1159→
  1160→  const renderCell = (value: unknown, colName: string) => {
  1161→    if (value === null || value === undefined || value === '') {
  1162→      return <span className="text-zinc-700">∅</span>
  1163→    }
  1164→    const type = typeByName.get(colName)
  1165→    if (typeof value === 'boolean' || type === 'boolean') {
  1166→      return (
  1167→        <span className={value ? 'text-amber-400' : 'text-zinc-500'}>{String(value)}</span>
  1168→      )
  1169→    }
  1170→    if (typeof value === 'number' || type === 'number') {
  1171→      const n = typeof value === 'number' ? value : Number(value)
  1172→      return <span className="tabular-nums">{Number.isNaN(n) ? String(value) : n.toLocaleString()}</span>
  1173→    }
  1174→    return String(value)
  1175→  }
  1176→
  1177→  return (
  1178→    <div className="rounded-md border border-zinc-800 bg-zinc-950/50">
  1179→      <div className="flex flex-wrap items-center justify-between gap-2 border-b border-zinc-800 px-3 py-2">
  1180→        <p className="flex items-center gap-1.5 font-mono text-[11px] text-zinc-400">
  1181→          <Braces className="h-3.5 w-3.5 text-emerald-400" aria-hidden />
  1182→          output → <span className="text-emerald-400">{outputName}</span>
  1183→          {result.run.outputVersionNumber !== null && (
  1184→            <span className="rounded border border-zinc-800 bg-zinc-900 px-1 text-[10px] text-zinc-500">
  1185→              v{result.run.outputVersionNumber}
  1186→            </span>
  1187→          )}
  1188→        </p>
  1189→        <p className="font-mono text-[10px] text-zinc-500">
  1190→          showing {rows.length} of {(result.run.rowCount ?? rows.length).toLocaleString()} rows ·{' '}
  1191→          {result.run.durationMs !== null ? `${result.run.durationMs}ms` : '—'}
  1192→        </p>
  1193→      </div>
  1194→      <div className="foundry-scroll max-h-80 overflow-auto lg:max-h-none">
  1195→        <Table>
  1196→          <TableHeader className="sticky top-0 z-10 bg-zinc-900">
  1197→            <TableRow className="border-zinc-800 hover:bg-transparent">
  1198→              {columns.map((col) => (
  1199→                <TableHead
  1200→                  key={col.name}
  1201→                  className={cn(
  1202→                    'h-8 whitespace-nowrap px-3 font-mono text-[11px] text-zinc-400',
  1203→                    col.type === 'number' && 'text-right',
  1204→                  )}
  1205→                >
  1206→                  {col.name}
  1207→                  <span className="ml-1 text-zinc-600">
  1208→                    {col.type === 'number' ? '#' : col.type === 'boolean' ? '⌥' : 'A'}
  1209→                  </span>
  1210→                </TableHead>
  1211→              ))}
  1212→            </TableRow>
  1213→          </TableHeader>
  1214→          <TableBody>
  1215→            {rows.map((row, i) => (
  1216→              <TableRow key={i} className="border-zinc-800/70 hover:bg-zinc-900/70">

  1230→          </TableBody>
  1231→        </Table>

  1233→    </div>
