   155→function NavLink({
   156→  item,
   157→  rail,
   158→  onNavigate,
   159→}: {
   160→  item: NavItem
   161→  rail: boolean
   162→  onNavigate?: () => void
   163→}) {
   164→  const pathname = usePathname()
   165→  const active = isActive(pathname, item.href)
   166→  const link = (
   167→    <Link
   168→      href={item.href}
   169→      onClick={onNavigate}
   170→      aria-current={active ? 'page' : undefined}
   171→      aria-label={rail ? item.label : undefined}
   172→      className={cn(
   173→        'group relative flex h-8 shrink-0 items-center rounded-md text-[13px] font-medium transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/50',
   174→        rail ? 'w-full justify-center px-0' : 'gap-2.5 px-2.5',
   175→        active
   176→          ? 'bg-emerald-500/[0.12] text-emerald-300'
   177→          : 'text-zinc-400 hover:bg-zinc-900 hover:text-zinc-100',
   178→      )}
   179→    >
   180→      {active && !rail && (
   181→        <span
   182→          className="absolute inset-y-[7px] left-0 w-[2.5px] rounded-full bg-emerald-400"
   183→          aria-hidden
   184→        />
   185→      )}
   186→      <item.icon
   187→        className={cn(
   188→          'size-4 shrink-0 transition-colors',
   189→          active
   190→            ? 'text-emerald-400'
   191→            : 'text-zinc-500 group-hover:text-zinc-300',
   192→        )}
   193→        aria-hidden
   194→      />
   195→      {!rail && <span className="truncate">{item.label}</span>}
   196→    </Link>
   197→  )
   198→
   199→  // Icon-only rail gets a hover tooltip; full sidebar labels speak for themselves.
   200→  if (!rail) return link
   201→  return (
   202→    <Tooltip>
   203→      <TooltipTrigger asChild>{link}</TooltipTrigger>
   204→      <TooltipContent
   205→        side="right"
   206→        sideOffset={10}
   207→        className="border-zinc-800 bg-zinc-900 px-2 py-1 text-xs text-zinc-200"
   208→      >
   209→        {item.label}
   210→      </TooltipContent>
   211→    </Tooltip>
   212→  )
   213→}

   479→/**
   480→ * Signed-in identity — avatar menu (task 7-c). Shows the real identity
   481→ * (name/email/role badge), a "Switch role (admin)" impersonation submenu for
   482→ * session admins, and Sign out (full cache invalidation so role-gated data
   483→ * downgrades back to demo level).
   484→ */
   485→function UserMenu({ session }: { session: SessionResponse }) {
   486→  const queryClient = useQueryClient()
   487→  const user: SessionUserInfo | null = session.session ?? null
   488→  // EFFECTIVE role (impersonation-aware); `user.role` stays the TRUE role.
   489→  const effectiveRole = user?.effectiveRole ?? session.role
   490→  const isAdminUser = user?.role === 'admin'
   491→  const impersonating = Boolean(user?.impersonating)
   492→
   493→  const logoutMutation = useMutation({
   494→    mutationFn: () => apiSend<{ ok: true }>('/api/auth/logout', 'POST'),
   495→    onSuccess: async () => {
   496→      // Demo-safe default: drop the simulator cookie to viewer so the
   497→      // anonymous session starts least-privileged (the bare no-cookie
   498→      // machine default would be admin). Best effort — ignore failures.
   499→      try {
   500→        await apiSend('/api/session/role', 'POST', { role: 'viewer' })
   501→      } catch {
   502→        /* demo default only — the refetch below settles the UI either way */
   503→      }
   504→      toast.success('Signed out', {
   505→        description: 'Demo mode — acting as viewer. Sign in to use a real identity.',
   506→      })
   507→      queryClient.invalidateQueries()
   508→    },
   509→    onError: (err) => toastApiError(err, 'Could not sign out'),
   510→  })
   511→
   512→  // Impersonation is REAL (session-carried): the server re-issues the session
   513→  // token with the effective role, so refetching ['session'] brings the truth
   514→  // back and every role-gated query re-resolves against the new role.
   515→  const switchRoleMutation = useMutation({
   516→    mutationFn: (next: FoundryRole) =>
   517→      apiSend<{ role: FoundryRole }>('/api/session/role', 'POST', { role: next }),
   518→    onSuccess: (res) => {
   519→      toast.success(
   520→        res.role === 'admin' ? 'Restored admin role' : `Acting as ${res.role}`,
   521→        {
   522→          description:
   523→            res.role === 'admin'
   524→              ? 'Full admin permissions are back.'
   525→              : 'API permissions now follow this role until you switch back or sign out.',
   526→        },
   527→      )
   528→      queryClient.invalidateQueries()
   529→    },
   530→    onError: (err) => toastApiError(err, 'Could not switch role'),
   531→  })
   532→
   533→  if (!user) return null
   534→  const initials = user.name
   535→    .split(/\s+/)
   536→    .filter(Boolean)
   537→    .map((w) => w[0])
   538→    .slice(0, 2)
   539→    .join('')
   540→    .toUpperCase()
   541→
   542→  return (
   543→    <DropdownMenu>
   544→      <DropdownMenuTrigger asChild>
   545→        <button
   546→          className="flex h-8 shrink-0 items-center gap-1.5 rounded-full border border-zinc-800 bg-zinc-900/60 py-1 pl-1 pr-1.5 transition-colors hover:border-zinc-700 hover:bg-zinc-800 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/50"
   547→          aria-label={`Account menu — ${user.name} (${effectiveRole}${impersonating ? ', impersonated' : ''})`}
   548→        >
   549→          <span
   550→            className="flex size-6 items-center justify-center rounded-full border border-emerald-500/40 bg-emerald-500/15 font-mono text-[9px] font-bold text-emerald-300"
   551→            aria-hidden
   552→          >
   553→            {initials}
   554→          </span>
   555→          <span className="hidden max-w-32 truncate text-xs font-medium text-zinc-200 md:inline">
   556→            {user.name}
   557→          </span>

   636→const SIDEBAR_KEY = 'mf.sidebar.collapsed'
   637→
   638→/**
   639→ * Rail/expanded preference, persisted in localStorage. Backed by
   640→ * useSyncExternalStore so the client picks up the stored value during
   641→ * hydration (no effect, no flash) and cross-tab updates stay in sync.
   642→ */
   643→function useSidebarCollapsed(): [boolean, (next: boolean) => void] {
   644→  const subscribe = useCallback((onStoreChange: () => void) => {
   645→    window.addEventListener('mf:sidebar', onStoreChange)
   646→    window.addEventListener('storage', onStoreChange)
   647→    return () => {
   648→      window.removeEventListener('mf:sidebar', onStoreChange)
   649→      window.removeEventListener('storage', onStoreChange)
   650→    }
   651→  }, [])
   652→  const collapsed = useSyncExternalStore(
   653→    subscribe,
   654→    () => window.localStorage.getItem(SIDEBAR_KEY) === '1',
   655→    () => false, // server snapshot: expanded
   656→  )
   657→  const setCollapsed = useCallback((next: boolean) => {
   658→    try {
   659→      window.localStorage.setItem(SIDEBAR_KEY, next ? '1' : '0')
   660→    } catch {
   661→      /* private mode etc. — preference just won't persist */
   662→    }
   663→    window.dispatchEvent(new Event('mf:sidebar'))
   664→  }, [])
   665→  return [collapsed, setCollapsed]
   666→}
