    55→    55→function useDebouncedValue<T>(value: T, delayMs: number): T {
    56→    56→  const [debounced, setDebounced] = useState(value)
    57→    57→  useEffect(() => {
    58→    58→    const t = setTimeout(() => setDebounced(value), delayMs)
    59→    59→    return () => clearTimeout(t)
    60→    60→  }, [value, delayMs])
    61→    61→  return debounced
    62→    62→}

    64→    64→/** Compact value renderer for object cells / property rows. */
    65→    65→function renderValue(value: unknown): React.ReactNode {
    66→    66→  if (value === null || value === undefined || value === '') {
    67→    67→    return <span className="text-muted-foreground/70">∅</span>
    68→    68→  }
    69→    69→  if (value === MASK_LITERAL) {
    70→    70→    return <span className="italic text-muted-foreground/80">{MASK_LITERAL}</span>
    71→    71→  }
    72→    72→  if (typeof value === 'boolean') {
    73→    73→    return <span className={value ? 'text-amber-600 dark:text-amber-400' : 'text-muted-foreground'}>{String(value)}</span>
    74→    74→  }
    75→    75→  if (typeof value === 'number') {
    76→    76→    return <span className="tabular-nums">{value.toLocaleString()}</span>
    77→    77→  }
    78→    78→  return String(value)
    79→    79→}
